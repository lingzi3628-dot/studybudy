   332→   332→function UsersTab() {
   333→   333→  const [users, setUsers] = useState<AdminUser[]>([]);
   334→   334→  const [loading, setLoading] = useState(true);
   335→   335→  const [error, setError] = useState<string | null>(null);
   336→   336→  const [q, setQ] = useState("");
   337→   337→  const [planFilter, setPlanFilter] = useState("all");
   338→   338→  const [roleFilter, setRoleFilter] = useState("all");
   339→   339→  const [page, setPage] = useState(1);
   340→   340→  const [totalPages, setTotalPages] = useState(1);
   341→   341→  const [editing, setEditing] = useState<AdminUser | null>(null);
   342→   342→
   343→   343→  const load = useCallback(async () => {
   344→   344→    setLoading(true);
   345→   345→    setError(null);
   346→   346→    try {
   347→   347→      const url = new URL("/api/admin/users", window.location.origin);
   348→   348→      url.searchParams.set("q", q);
   349→   349→      url.searchParams.set("plan", planFilter);
   350→   350→      url.searchParams.set("role", roleFilter);
   351→   351→      url.searchParams.set("page", String(page));
   352→   352→      const r = await fetch(url);
   353→   353→      if (!r.ok) throw new Error(`HTTP ${r.status}`);
   354→   354→      const d = await r.json();
   355→   355→      setUsers(d.users);
   356→   356→      setTotalPages(d.totalPages);
   357→   357→    } catch (e: any) {
   358→   358→      setError(e?.message ?? "Failed to load users");
   359→   359→    } finally {
   360→   360→      setLoading(false);
   361→   361→    }
   362→   362→  }, [q, planFilter, roleFilter, page]);
   363→   363→
   364→   364→  useEffect(() => {
   365→   365→    load();
   366→   366→  }, [load]);
   367→   367→
   368→   368→  const updateUser = async (id: string, body: { plan?: string; role?: string; banned?: boolean }) => {
   369→   369→    try {
   370→   370→      const r = await fetch(`/api/admin/users/${id}`, {
   371→   371→        method: "PUT",
   372→   372→        headers: { "Content-Type": "application/json" },
   373→   373→        body: JSON.stringify(body),
   374→   374→      });
   375→   375→      if (!r.ok) {
   376→   376→        const d = await r.json().catch(() => ({}));
   377→   377→        throw new Error(d.error ?? `HTTP ${r.status}`);
   378→   378→      }
   379→   379→      await load();
   380→   380→    } catch (e: any) {
   381→   381→      setError(e?.message ?? "Update failed");
   382→   382→    }
   383→   383→  };
   384→   384→
   385→   385→  const deleteUser = async (id: string) => {
   386→   386→    if (!confirm("Delete this user and all their data? This cannot be undone.")) return;
   387→   387→    try {
   388→   388→      const r = await fetch(`/api/admin/users/${id}`, { method: "DELETE" });
   389→   389→      if (!r.ok) throw new Error(`HTTP ${r.status}`);
   390→   390→      await load();
   391→   391→    } catch (e: any) {
   392→   392→      setError(e?.message ?? "Delete failed");
   393→   393→    }
   394→   394→  };
   395→   395→
   396→   396→  if (loading) return <Spinner label="Loading users…" />;
   397→   397→  if (error) return <ErrorBox message={error} />;
   398→   398→
   399→   399→  return (
   400→   400→    <div className="space-y-3">
   401→   401→      {/* search + filters */}
   402→   402→      <div className="flex gap-2">
   403→   403→        <div className="flex-1 flex items-center gap-2 bg-white border border-gray-200 rounded-full px-3 h-10">
   404→   404→          <Search className="w-4 h-4 text-gray-400" />
   405→   405→          <input
   406→   406→            value={q}
   407→   407→            onChange={(e) => { setQ(e.target.value); setPage(1); }}
   408→   408→            placeholder="Search by email, name, or ID…"
   409→   409→            className="flex-1 bg-transparent outline-none text-sm"
   410→   410→          />
   411→   411→        </div>
   412→   412→        <select value={planFilter} onChange={(e) => { setPlanFilter(e.target.value); setPage(1); }} className="bg-white border border-gray-200 rounded-full px-3 h-10 text-xs">
   413→   413→          <option value="all">All plans</option>
   414→   414→          <option value="free">Free</option>
   415→   415→          <option value="pro">Pro</option>
   416→   416→        </select>
