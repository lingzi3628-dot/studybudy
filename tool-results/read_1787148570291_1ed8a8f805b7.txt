   134→                      if (o.key === "graph") {
   135→                        closeCreate();
   136→                        setScreen("graph");
   137→                        return;
   138→                      }
   139→                      if (o.key === "tutor") {
   140→                        closeCreate();
   141→                        setScreen("tutor");
   142→                        return;
   143→                      }
   144→                      if (o.key === "path") {
   145→                        closeCreate();
   146→                        setScreen("path");
   147→                        return;
   148→                      }
   149→                      openCreate(o.key);
   150→                      setStep("input");
   151→                    }}
   152→                    className="w-full flex items-center gap-4 p-3.5 rounded-2xl border border-gray-200 hover:border-indigo-300 hover:bg-indigo-50/40 transition text-left"
   153→                  >
   154→                    <span className={`w-10 h-10 rounded-full flex items-center justify-center ${o.color}`}>
   155→                      <Icon className="w-5 h-5" />
   156→                    </span>
   157→                    <div className="flex-1 min-w-0">
   158→                      <p className="text-sm font-semibold text-gray-900">{o.label}</p>
   159→                      <p className="text-xs text-gray-500 truncate">{o.desc}</p>
   160→                    </div>
   161→                  </button>
   162→                );
   163→              })}
   164→            </div>
   165→          </div>
   166→        </div>
   167→      </div>
   168→    );
   169→  }
   170→
   171→  // ============ "success" step ============
   172→  if (step === "success" && savedSetId) {
   173→    return (
   174→      <div className="fixed inset-0 z-50 flex items-end sm:items-center justify-center">
   175→        <div className="absolute inset-0 bg-black/40" onClick={closeCreate} aria-hidden />
   176→        <div className="relative w-full max-w-md bg-white rounded-t-3xl sm:rounded-3xl shadow-2xl animate-in slide-in-from-bottom-4">
   177→          <div className="p-6 text-center">
   178→            <div className="w-12 h-12 mx-auto rounded-full bg-emerald-100 text-emerald-600 flex items-center justify-center">
   179→              <Check className="w-6 h-6" />
   180→            </div>
   181→            <h3 className="mt-3 text-base font-semibold text-gray-900">Study set saved!</h3>
   182→            <p className="mt-1 text-sm text-gray-500">
   183→              {editedFlashcards.length + editedMcqs.length} cards ready to study.
   184→            </p>
   185→            <div className="mt-4 space-y-2">
   186→              <button
   187→                onClick={() => {
   188→                  setActiveStudySetId(savedSetId);
   189→                  closeCreate();
   190→                  setScreen("quiz");
   191→                }}
   192→                className="w-full h-11 rounded-full bg-indigo-600 text-white font-semibold shadow-md hover:bg-indigo-700"
   193→              >
   194→                Start studying now
   195→              </button>
   196→              <button
   197→                onClick={() => openCreate(null)}
   198→                className="w-full h-10 rounded-full text-gray-500 text-sm hover:bg-gray-100"
   199→              >
   200→                Create another
   201→              </button>
   202→            </div>
   203→          </div>
   204→        </div>
   205→      </div>
   206→    );
   207→  }
   208→
   209→  // ============ "generating" / "saving" step ============
   210→  if (step === "generating" || step === "saving") {
   211→    return (
   212→      <div className="fixed inset-0 z-50 flex items-end sm:items-center justify-center">
   213→        <div className="absolute inset-0 bg-black/40" aria-hidden />
   214→        <div className="relative w-full max-w-md bg-white rounded-t-3xl sm:rounded-3xl shadow-2xl p-6 text-center">
   215→          <Loader2 className="w-8 h-8 mx-auto text-indigo-600 animate-spin" />
   216→          <h3 className="mt-3 text-base font-semibold text-gray-900">
   217→            {step === "generating" ? "Generating with AI…" : "Saving to database…"}
   218→          </h3>
   219→          <p className="mt-1 text-xs text-gray-500">
   220→            {step === "generating"
   221→              ? `Asking the AI to write ${numFlashcards > 0 ? `${numFlashcards} flashcard${numFlashcards === 1 ? "" : "s"}` : ""}${numFlashcards > 0 && numMCQs > 0 ? " and " : ""}${numMCQs > 0 ? `${numMCQs} MCQ${numMCQs === 1 ? "" : "s"}` : ""}.`
   222→              : "Writing cards to your Neon Postgres database."}
   223→          </p>
   224→        </div>
   225→      </div>
   226→    );
   227→  }
   228→
   229→  // ============ "preview" step — edit/delete cards then save ============
   230→  if (step === "preview" && generated) {
   231→    return (
   232→      <div className="fixed inset-0 z-50 flex items-end sm:items-center justify-center">
   233→        <div className="absolute inset-0 bg-black/40" onClick={closeCreate} aria-hidden />
   234→        <div className="relative w-full max-w-2xl bg-white rounded-t-3xl sm:rounded-3xl shadow-2xl max-h-[92vh] overflow-y-auto">
   235→          <div className="sticky top-0 bg-white px-5 pt-5 pb-3 border-b border-gray-100 flex items-center justify-between">
   236→            <div className="flex items-center gap-2 min-w-0">
   237→              <button
   238→                onClick={() => {
   239→                  setStep("input");
   240→                  setGenerated(null);
   241→                }}
   242→                aria-label="Back"
   243→                className="w-8 h-8 rounded-full bg-gray-100 hover:bg-gray-200 flex items-center justify-center text-gray-600 flex-shrink-0"
   244→              >
   245→                <ChevronLeft className="w-4 h-4" />
   246→              </button>
   247→              <h2 className="text-lg font-bold text-gray-900 truncate">
   248→                Preview &amp; Edit
   249→              </h2>
   250→            </div>
   251→            <button
   252→              onClick={closeCreate}
   253→              aria-label="Close"
   254→              className="w-8 h-8 rounded-full bg-gray-100 hover:bg-gray-200 flex items-center justify-center text-gray-600"
   255→            >
   256→              <X className="w-4 h-4" />
   257→            </button>
   258→          </div>
   259→
   260→          <div className="p-5 space-y-5">
   261→            <div className="grid grid-cols-2 gap-2 text-xs">
   262→              <div>
   263→                <p className="text-gray-500">Flashcards</p>
   264→                <p className="font-semibold text-gray-900">{editedFlashcards.length}</p>
   265→              </div>
   266→              <div>
   267→                <p className="text-gray-500">MCQs</p>
   268→                <p className="font-semibold text-gray-900">{editedMcqs.length}</p>
   269→              </div>
   270→            </div>
   271→
   272→            {editedFlashcards.length > 0 && (
   273→              <div>
   274→                <h3 className="text-xs font-semibold uppercase tracking-wide text-gray-500 mb-2">Flashcards</h3>
   275→                <div className="space-y-2">
   276→                  {editedFlashcards.map((c, i) => (
   277→                    <div key={i} className="rounded-2xl border border-gray-200 p-3">
   278→                      <div className="flex items-start gap-2">
   279→                        <span className="w-6 h-6 rounded-full bg-indigo-50 text-indigo-600 flex items-center justify-center text-[10px] font-bold flex-shrink-0">
   280→                          {i + 1}
   281→                        </span>
   282→                        <div className="flex-1 space-y-1">
   283→                          <input
   284→                            value={c.front}
   285→                            onChange={(e) => {
   286→                              setEditedFlashcards((arr) => arr.map((x, j) => j === i ? { ...x, front: e.target.value } : x));
   287→                            }}
   288→                            className="w-full text-sm font-medium text-gray-900 bg-transparent outline-none border-b border-transparent focus:border-indigo-300"
   289→                          />
   290→                          <input
   291→                            value={c.back}
   292→                            onChange={(e) => {
   293→                              setEditedFlashcards((arr) => arr.map((x, j) => j === i ? { ...x, back: e.target.value } : x));
   294→                            }}
   295→                            className="w-full text-xs text-gray-600 bg-transparent outline-none border-b border-transparent focus:border-indigo-300"
   296→                          />
   297→                        </div>
   298→                        <button
   299→                          onClick={() => setEditedFlashcards((arr) => arr.filter((_, j) => j !== i))}
   300→                          className="w-7 h-7 rounded-full hover:bg-rose-50 text-rose-500 flex items-center justify-center flex-shrink-0"
   301→                          aria-label="Delete"
   302→                        >
   303→                          <Trash2 className="w-3.5 h-3.5" />
   304→                        </button>
   305→                      </div>
   306→                    </div>
   307→                  ))}
   308→                </div>
   309→              </div>
   310→            )}
   311→
   312→            {editedMcqs.length > 0 && (
   313→              <div>
   314→                <h3 className="text-xs font-semibold uppercase tracking-wide text-gray-500 mb-2">Multiple-choice questions</h3>
   315→                <div className="space-y-2">
   316→                  {editedMcqs.map((c, i) => (
   317→                    <div key={i} className="rounded-2xl border border-gray-200 p-3">
   318→                      <div className="flex items-start gap-2">
   319→                        <span className="w-6 h-6 rounded-full bg-emerald-50 text-emerald-600 flex items-center justify-center text-[10px] font-bold flex-shrink-0">
   320→                          {i + 1}
   321→                        </span>
   322→                        <div className="flex-1 space-y-1.5">
   323→                          <input
   324→                            value={c.question}
   325→                            onChange={(e) => {
   326→                              setEditedMcqs((arr) => arr.map((x, j) => j === i ? { ...x, question: e.target.value } : x));
   327→                            }}
   328→                            className="w-full text-sm font-medium text-gray-900 bg-transparent outline-none border-b border-transparent focus:border-emerald-300"
   329→                          />
   330→                          {c.options.map((opt, j) => (
   331→                            <div key={j} className="flex items-center gap-1.5">
   332→                              <input
   333→                                type="radio"
   334→                                checked={c.correct_index === j}
   335→                                onChange={() => {
   336→                                  setEditedMcqs((arr) => arr.map((x, k) => k === i ? { ...x, correct_index: j } : x));
   337→                                }}
   338→                                className="w-3 h-3 accent-emerald-500"
   339→                              />
   340→                              <input
   341→                                value={opt}
   342→                                onChange={(e) => {
   343→                                  setEditedMcqs((arr) => arr.map((x, k) => k === i ? { ...x, options: x.options.map((o, l) => l === j ? e.target.value : o) } : x));
   344→                                }}
   345→                                className="flex-1 text-xs text-gray-700 bg-transparent outline-none border-b border-transparent focus:border-emerald-300"
   346→                              />
   347→                            </div>
   348→                          ))}
   349→                          <input
   350→                            value={c.explanation}
   351→                            onChange={(e) => {
   352→                              setEditedMcqs((arr) => arr.map((x, j) => j === i ? { ...x, explanation: e.target.value } : x));
   353→                            }}
   354→                            className="w-full text-xs italic text-gray-500 bg-transparent outline-none border-b border-transparent focus:border-emerald-300"
   355→                          />
   356→                        </div>
   357→                        <button
   358→                          onClick={() => setEditedMcqs((arr) => arr.filter((_, j) => j !== i))}
   359→                          className="w-7 h-7 rounded-full hover:bg-rose-50 text-rose-500 flex items-center justify-center flex-shrink-0"
   360→                          aria-label="Delete"
   361→                        >
   362→                          <Trash2 className="w-3.5 h-3.5" />
   363→                        </button>
   364→                      </div>
   365→                    </div>
   366→                  ))}
   367→                </div>
   368→              </div>
   369→            )}
   370→
   371→            {error && (
   372→              <div className="flex items-start gap-2 p-3 rounded-xl bg-rose-50 text-rose-700 text-xs">
   373→                <AlertCircle className="w-4 h-4 flex-shrink-0 mt-0.5" />
   374→                <span>{error}</span>
   375→              </div>
   376→            )}
   377→
   378→            <button
   379→              onClick={async () => {
   380→                setStep("saving");
   381→                setError(null);
   382→                try {
   383→                  // Pass the edited cards directly — no AI re-generation needed
   384→                  const cardsToSave = [
   385→                    ...editedFlashcards.map((c) => ({
   386→                      cardType: "flashcard" as const,
   387→                      front: c.front,
   388→                      back: c.back,
   389→                      question: null,
   390→                      options: null,
   391→                      correctIndex: null,
   392→                      explanation: null,
   393→                    })),
   394→                    ...editedMcqs.map((c) => ({
   395→                      cardType: "mcq" as const,
   396→                      front: null,
   397→                      back: null,
   398→                      question: c.question,
   399→                      options: c.options,
   400→                      correctIndex: c.correct_index,
   401→                      explanation: c.explanation,
   402→                    })),
   403→                  ];
   404→
   405→                  const res = await api.createStudySet({
   406→                    title: title || "Untitled set",
   407→                    sourceType: createOption === "upload" ? "pdf" : "text",
   408→                    sourceText: text,
   409→                    subject: subject || undefined,
   410→                    topic: topic || undefined,
   411→                    generate: false,
   412→                    cards: cardsToSave,
   413→                  });
   414→                  setSavedSetId(res.studySet.id);
   415→                  setStep("success");
   416→                } catch (e: any) {
   417→                  setError(e?.message ?? "Save failed");
   418→                  setStep("preview");
   419→                }
   420→              }}
   421→              disabled={busy || (editedFlashcards.length === 0 && editedMcqs.length === 0)}
   422→              className="w-full h-12 rounded-full bg-indigo-600 text-white font-semibold shadow-md hover:bg-indigo-700 transition disabled:opacity-50 flex items-center justify-center gap-1.5"
   423→            >
   424→              <Sparkles className="w-4 h-4" /> Save Study Set
   425→            </button>
   426→          </div>
   427→        </div>
   428→      </div>
   429→    );
   430→  }
   431→
   432→  // ============ "input" step — option-specific input form ============
   433→  // Determine config per option
   434→  const cfg = createOption
   435→    ? {
   436→        upload: { title: "Upload a file", icon: UploadCloud, color: "bg-indigo-50 text-indigo-600" },
   437→        paste: { title: "Paste your text", icon: Clipboard, color: "bg-violet-50 text-violet-600" },
   438→        flashcards: { title: "Generate flashcards", icon: Layers, color: "bg-amber-50 text-amber-600" },
   439→        quiz: { title: "Generate a quiz", icon: ListChecks, color: "bg-emerald-50 text-emerald-600" },
   440→      }[createOption as "upload" | "paste" | "flashcards" | "quiz"]
   441→    : null;
   442→
   443→  if (!cfg) {
   444→    // unknown option — close
   445→    return null;
   446→  }
   447→
   448→  const Icon = cfg.icon;
   449→
   450→  const handleGenerate = async () => {
   451→    if (createOption === "upload" && !text.trim()) {
   452→      setError("Please upload a file and extract text first.");
   453→      return;
   454→    }
   455→    if ((createOption === "paste" || createOption === "flashcards" || createOption === "quiz") && !text.trim()) {
   456→      setError("Please enter some text or topic first.");
   457→      return;
   458→    }
   459→    setStep("generating");
   460→    setError(null);
   461→    try {
   462→      const result = await api.generateCards({
   463→        text,
   464→        numFlashcards,
   465→        numMCQs,
   466→        subject: subject || undefined,
   467→        topic: topic || undefined,
   468→      });
   469→      setGenerated(result);
   470→      setEditedFlashcards(result.flashcards);
   471→      setEditedMcqs(result.mcqs);
   472→      setStep("preview");
   473→    } catch (e: any) {
   474→      setError(e?.message ?? "Generation failed");
   475→      setStep("input");
   476→    }
   477→  };
   478→
   479→  const handleFilePick = async (f: File) => {
   480→    setFile(f);
   481→    setExtracting(true);
   482→    setError(null);
   483→    setText("");
   484→    try {
   485→      const r = await api.extractFile(f);
   486→      setText(r.text);
   487→      if (!title) setTitle(f.name.replace(/\.[^.]+$/, ""));
   488→    } catch (e: any) {
   489→      setError(e?.message ?? "Extraction failed");
   490→    } finally {
   491→      setExtracting(false);
   492→    }
   493→  };
   494→
   495→  return (
   496→    <div className="fixed inset-0 z-50 flex items-end sm:items-center justify-center">
   497→      <div className="absolute inset-0 bg-black/40" onClick={closeCreate} aria-hidden />
   498→      <div className="relative w-full max-w-md bg-white rounded-t-3xl sm:rounded-3xl shadow-2xl max-h-[92vh] overflow-y-auto animate-in slide-in-from-bottom-4">
   499→        <div className="sticky top-0 bg-white px-5 pt-5 pb-3 border-b border-gray-100 flex items-center justify-between">
   500→          <div className="flex items-center gap-2 min-w-0">
   501→            <button
   502→              onClick={() => openCreate(null)}
   503→              aria-label="Back"
   504→              className="w-8 h-8 rounded-full bg-gray-100 hover:bg-gray-200 flex items-center justify-center text-gray-600 flex-shrink-0"
   505→            >
   506→              <ChevronLeft className="w-4 h-4" />
   507→            </button>
   508→            <h2 className="text-lg font-bold text-gray-900 truncate">{cfg.title}</h2>
   509→          </div>
   510→          <button
   511→            onClick={closeCreate}
   512→            aria-label="Close"
   513→            className="w-8 h-8 rounded-full bg-gray-100 hover:bg-gray-200 flex items-center justify-center text-gray-600"
   514→          >
   515→            <X className="w-4 h-4" />
   516→          </button>
   517→        </div>
   518→
   519→        <div className="p-5 space-y-3">
   520→          {/* File upload for "upload" option */}
   521→          {createOption === "upload" && (
   522→            <div className="rounded-2xl border-2 border-dashed border-gray-200 p-6 text-center">
   523→              <input
   524→                type="file"
   525→                accept=".pdf,.txt,.md,.markdown,.csv,application/pdf,text/plain"
   526→                onChange={(e) => {
   527→                  const f = e.target.files?.[0];
   528→                  if (f) handleFilePick(f);
   529→                }}
   530→                className="hidden"
   531→                id="file-input"
   532→              />
   533→              <label htmlFor="file-input" className="cursor-pointer inline-flex flex-col items-center gap-2">
   534→                <span className={`w-12 h-12 rounded-full flex items-center justify-center ${cfg.color}`}>
   535→                  {extracting ? <Loader2 className="w-5 h-5 animate-spin" /> : <Icon className="w-5 h-5" />}
   536→                </span>
   537→                <span className="text-sm font-medium text-gray-900">
   538→                  {extracting ? "Extracting text…" : file ? file.name : "Tap to pick a PDF or .txt"}
   539→                </span>
   540→                <span className="text-xs text-gray-400">PDF or .txt up to 5 MB</span>
   541→              </label>
   542→            </div>
   543→          )}
   544→
   545→          {/* Title */}
   546→          <div>
   547→            <label className="text-xs font-semibold uppercase tracking-wide text-gray-500">Title</label>
   548→            <input
   549→              value={title}
   550→              onChange={(e) => setTitle(e.target.value)}
   551→              placeholder={file ? file.name.replace(/\.[^.]+$/, "") : "Auto-generated if empty"}
   552→              className="mt-1.5 w-full p-3 rounded-2xl border border-gray-200 text-sm outline-none focus:border-indigo-400 focus:ring-2 focus:ring-indigo-100"
   553→            />
   554→          </div>
   555→
   556→          {/* Subject + Topic */}
   557→          <div className="grid grid-cols-2 gap-2">
   558→            <div>
   559→              <label className="text-xs font-semibold uppercase tracking-wide text-gray-500">Subject</label>
   560→              <input
   561→                value={subject}
   562→                onChange={(e) => setSubject(e.target.value)}
   563→                placeholder="e.g. Science"
   564→                className="mt-1.5 w-full p-3 rounded-2xl border border-gray-200 text-sm outline-none focus:border-indigo-400 focus:ring-2 focus:ring-indigo-100"
   565→              />
   566→            </div>
   567→            <div>
   568→              <label className="text-xs font-semibold uppercase tracking-wide text-gray-500">Topic</label>
   569→              <input
   570→                value={topic}
   571→                onChange={(e) => setTopic(e.target.value)}
   572→                placeholder="e.g. Photosynthesis"
   573→                className="mt-1.5 w-full p-3 rounded-2xl border border-gray-200 text-sm outline-none focus:border-indigo-400 focus:ring-2 focus:ring-indigo-100"
   574→              />
   575→            </div>
   576→          </div>
   577→
   578→          {/* Counts — show for flashcards & quiz */}
   579→          {(createOption === "flashcards" || createOption === "quiz") && (
   580→            <div className="grid grid-cols-2 gap-2">
   581→              <div>
   582→                <label className="text-xs font-semibold uppercase tracking-wide text-gray-500">
   583→                  Flashcards
   584→                </label>
   585→                <input
   586→                  type="number"
   587→                  min={0}
   588→                  max={12}
   589→                  value={numFlashcards}
   590→                  onChange={(e) => setNumFlashcards(Math.max(0, Math.min(12, Number(e.target.value) || 0)))}
   591→                  className="mt-1.5 w-full p-3 rounded-2xl border border-gray-200 text-sm outline-none focus:border-indigo-400 focus:ring-2 focus:ring-indigo-100"
   592→                />
   593→              </div>
   594→              <div>
   595→                <label className="text-xs font-semibold uppercase tracking-wide text-gray-500">MCQs</label>
   596→                <input
   597→                  type="number"
   598→                  min={0}
   599→                  max={12}
   600→                  value={numMCQs}
   601→                  onChange={(e) => setNumMCQs(Math.max(0, Math.min(12, Number(e.target.value) || 0)))}
   602→                  className="mt-1.5 w-full p-3 rounded-2xl border border-gray-200 text-sm outline-none focus:border-indigo-400 focus:ring-2 focus:ring-indigo-100"
   603→                />
   604→              </div>
   605→            </div>
   606→          )}
   607→
   608→          {/* Textarea (paste / upload extracted text / flashcards source / quiz source) */}
   609→          {createOption !== "upload" && (
   610→            <div>
   611→              <label className="text-xs font-semibold uppercase tracking-wide text-gray-500">
   612→                {createOption === "flashcards" || createOption === "quiz"
   613→                  ? "Topic description or source text"
   614→                  : "Content"}
   615→              </label>
   616→              <textarea
   617→                value={text}
   618→                onChange={(e) => setText(e.target.value)}
   619→                rows={6}
   620→                placeholder={
   621→                  createOption === "flashcards" || createOption === "quiz"
   622→                    ? "e.g. Explain photosynthesis including the chemical equation and key organelles."
   623→                    : "Paste your notes, article, or text here..."
   624→                }
   625→                className="mt-1.5 w-full p-3 rounded-2xl border border-gray-200 text-sm outline-none focus:border-indigo-400 focus:ring-2 focus:ring-indigo-100 resize-none"
   626→              />
   627→              <p className="mt-1 text-[11px] text-gray-400">
   628→                {text.length} chars
   629→              </p>
   630→            </div>
   631→          )}
   632→
   633→          {/* For upload — also show extracted text */}
   634→          {createOption === "upload" && text && (
   635→            <div>
   636→              <label className="text-xs font-semibold uppercase tracking-wide text-gray-500 flex items-center justify-between">
   637→                <span>Extracted text (editable)</span>
   638→                <span className="text-[11px] font-normal text-gray-400">{text.length} chars</span>
   639→              </label>
   640→              <textarea
   641→                value={text}
   642→                onChange={(e) => setText(e.target.value)}
   643→                rows={6}
   644→                placeholder="Extracted text will appear here..."
   645→                className="mt-1.5 w-full p-3 rounded-2xl border border-gray-200 text-sm outline-none focus:border-indigo-400 focus:ring-2 focus:ring-indigo-100 resize-none font-mono"
   646→              />
   647→            </div>
   648→          )}
   649→
   650→          {error && (
   651→            <div className="flex items-start gap-2 p-3 rounded-xl bg-rose-50 text-rose-700 text-xs">
   652→              <AlertCircle className="w-4 h-4 flex-shrink-0 mt-0.5" />
   653→              <span>{error}</span>
   654→            </div>
   655→          )}
   656→
   657→          <div className="flex items-center gap-2 p-3 rounded-xl bg-amber-50 text-amber-700 text-xs">
   658→            <Sparkles className="w-4 h-4 flex-shrink-0" />
   659→            <span>
   660→              Generates {numFlashcards > 0 ? `${numFlashcards} flashcards` : ""}
   661→              {numFlashcards > 0 && numMCQs > 0 ? " + " : ""}
   662→              {numMCQs > 0 ? `${numMCQs} MCQs` : ""} using AI · uses 1 daily call
   663→            </span>
   664→          </div>
   665→
   666→          <button
   667→            onClick={handleGenerate}
   668→            disabled={busy || !text.trim()}
   669→            className="w-full h-12 rounded-full bg-indigo-600 text-white font-semibold shadow-md hover:bg-indigo-700 transition disabled:opacity-50 flex items-center justify-center gap-1.5"
   670→          >
   671→            <Sparkles className="w-4 h-4" /> Generate
   672→          </button>
   673→
   674→          <button
   675→            onClick={() => openCreate(null)}
   676→            className="w-full h-10 rounded-full text-gray-500 text-sm hover:bg-gray-100"
   677→          >
   678→            Back to options
   679→          </button>
   680→        </div>
   681→      </div>
   682→    </div>
   683→  );
   684→}
