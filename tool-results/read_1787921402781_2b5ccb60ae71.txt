    69→    69→// CSVDownloadButton — small button that downloads rows as a CSV file
    70→    70→// (opens in Excel, Google Sheets, LibreOffice)
    71→    71→function CSVDownloadButton({
    72→    72→  headers,
    73→    73→  rows,
    74→    74→  downloadName = "table.csv",
    75→    75→}: {
    76→    76→  headers: string[];
    77→    77→  rows: (string | number)[][];
    78→    78→  downloadName?: string;
    79→    79→}) {
    80→    80→  const escapeCSV = (val: string | number) => {
    81→    81→    const s = String(val ?? "");
    82→    82→    if (/[",\n]/.test(s)) return `"${s.replace(/"/g, '""')}"`;
    83→    83→    return s;
    84→    84→  };
    85→    85→  const csvText = [headers, ...rows].map((r) => r.map(escapeCSV).join(",")).join("\n");
    86→    86→
    87→    87→  const download = () => {
    88→    88→    const blob = new Blob([csvText], { type: "text/csv;charset=utf-8" });
    89→    89→    const url = URL.createObjectURL(blob);
    90→    90→    const a = document.createElement("a");
    91→    91→    a.href = url;
    92→    92→    a.download = downloadName.endsWith(".csv") ? downloadName : `${downloadName}.csv`;
    93→    93→    document.body.appendChild(a);
    94→    94→    a.click();
    95→    95→    document.body.removeChild(a);
    96→    96→    URL.revokeObjectURL(url);
    97→    97→  };
    98→    98→
    99→    99→  return (
   100→   100→    <button
   101→   101→      onClick={download}
   102→   102→      className="text-[10px] text-gray-500 hover:text-emerald-700 flex items-center gap-0.5"
   103→   103→      title="Download as CSV (opens in Excel)"
   104→   104→    >
   105→   105→      📄 CSV
   106→   106→    </button>
   107→   107→  );
   108→   108→}

   113→   113→// A modular dispatcher that renders many different kinds of graphs as

   116→   116→// Supported types:
   117→   117→//   1. function       — y = f(x) line plot (math)

   119→   119→//   3. bar            — vertical bar chart with categories
   120→   120→//   4. histogram      — like bar but no gaps, for frequency distributions
   121→   121→//   5. pie            — pie chart with labeled slices
   122→   122→//   6. venn           — 2 or 3-set Venn diagram (probability/sets)
   123→   123→//   7. numberline     — number line with shaded ranges (inequalities)

   225→   225→// =====================================================================
   226→   226→// 1. Function plot (y = f(x))
   227→   227→// =====================================================================
   228→   228→function FunctionSVG({ spec }: { spec: any }) {
   229→   229→  const expr = spec.expr || "x^2";
   230→   230→  const xRange: [number, number] = spec.xRange ?? [-5, 5];
   231→   231→  const yRange: [number, number] = spec.yRange ?? [-25, 25];
   232→   232→  const title = spec.title || `y = ${expr}`;
   233→   233→  const xLabel = spec.xLabel ?? "x";
   234→   234→  const yLabel = spec.yLabel ?? "y";
   235→   235→  const width = 480;
   236→   236→  const height = 320;
   237→   237→  const padding = 40;
   238→   238→
   239→   239→  const evaluate = (x: number): number | null => {
   240→   240→    try {
   241→   241→      let safeExpr = expr
   242→   242→        .replace(/\^/g, "**")
   243→   243→        .replace(/\bpi\b/gi, "Math.PI")
   244→   244→        .replace(/\be\b/g, "Math.E")
   245→   245→        .replace(/\bsin\(/g, "Math.sin(")
   246→   246→        .replace(/\bcos\(/g, "Math.cos(")
   247→   247→        .replace(/\btan\(/g, "Math.tan(")
   248→   248→        .replace(/\bsqrt\(/g, "Math.sqrt(")
   249→   249→        .replace(/\blog\(/g, "Math.log(")
   250→   250→        .replace(/\bexp\(/g, "Math.exp(")
   251→   251→        .replace(/\babs\(/g, "Math.abs(")
   252→   252→        .replace(/\bx\b/g, String(x));
   253→   253→      // eslint-disable-next-line no-new-func
   254→   254→      const fn = new Function("Math", `"use strict"; return (${safeExpr});`);
   255→   255→      const result = fn(Math);
   256→   256→      return typeof result === "number" && isFinite(result) ? result : null;
   257→   257→    } catch {
   258→   258→      return null;
   259→   259→    }
   260→   260→  };
   261→   261→
   262→   262→  const points: Array<{ x: number; y: number }> = [];
   263→   263→  const samples = 100;
   264→   264→  for (let i = 0; i <= samples; i++) {
   265→   265→    const x = xRange[0] + ((xRange[1] - xRange[0]) * i) / samples;
   266→   266→    const y = evaluate(x);
   267→   267→    if (y !== null) points.push({ x, y });
   268→   268→  }
   269→   269→
   270→   270→  const toSvgX = (x: number) => padding + ((x - xRange[0]) / (xRange[1] - xRange[0])) * (width - 2 * padding);
   271→   271→  const toSvgY = (y: number) =>
   272→   272→    height - padding - ((y - yRange[0]) / (yRange[1] - yRange[0])) * (height - 2 * padding);
   273→   273→
   274→   274→  const path = points
   275→   275→    .filter((p) => p.y >= yRange[0] && p.y <= yRange[1])
   276→   276→    .map((p, i) => `${i === 0 ? "M" : "L"} ${toSvgX(p.x).toFixed(2)} ${toSvgY(p.y).toFixed(2)}`)
   277→   277→    .join(" ");
   278→   278→
   279→   279→  const xAxisY = yRange[0] <= 0 && yRange[1] >= 0 ? toSvgY(0) : height - padding;
   280→   280→  const yAxisX = xRange[0] <= 0 && xRange[1] >= 0 ? toSvgX(0) : padding;
   281→   281→
   282→   282→  const xTicks: ReactElement[] = [];
   283→   283→  const tickCount = 5;
   284→   284→  for (let i = 0; i <= tickCount; i++) {
   285→   285→    const xv = xRange[0] + ((xRange[1] - xRange[0]) * i) / tickCount;
   286→   286→    xTicks.push(
   287→   287→      <g key={`xt-${i}`}>
   288→   288→        <line x1={toSvgX(xv)} y1={xAxisY - 4} x2={toSvgX(xv)} y2={xAxisY + 4} stroke="#9CA3AF" strokeWidth={1} />
   289→   289→        <text x={toSvgX(xv)} y={xAxisY + 16} fontSize={10} fill="#6B7280" textAnchor="middle">
   290→   290→          {xv.toFixed(1)}
   291→   291→        </text>
   292→   292→      </g>
   293→   293→    );
   294→   294→  }
   295→   295→
   296→   296→  const yTicks: ReactElement[] = [];
   297→   297→  for (let i = 0; i <= tickCount; i++) {
   298→   298→    const yv = yRange[0] + ((yRange[1] - yRange[0]) * i) / tickCount;
   299→   299→    yTicks.push(
   300→   300→      <g key={`yt-${i}`}>
   301→   301→        <line x1={yAxisX - 4} y1={toSvgY(yv)} x2={yAxisX + 4} y2={toSvgY(yv)} stroke="#9CA3AF" strokeWidth={1} />
   302→   302→        <text x={yAxisX - 8} y={toSvgY(yv) + 3} fontSize={10} fill="#6B7280" textAnchor="end">
   303→   303→          {yv.toFixed(0)}
   304→   304→        </text>
   305→   305→      </g>
   306→   306→    );
   307→   307→  }
   308→   308→
   309→   309→  return (
   310→   310→    <div>
   311→   311→      <p className="text-xs font-semibold text-gray-700 mb-1">{title}</p>
   312→   312→      <svg viewBox={`0 0 ${width} ${height}`} className="w-full h-auto bg-gray-50 rounded-lg">
   313→   313→        {xTicks}
   314→   314→        {yTicks}
   315→   315→        <line x1={padding} y1={xAxisY} x2={width - padding} y2={xAxisY} stroke="#374151" strokeWidth={1.5} />
   316→   316→        <line x1={yAxisX} y1={padding} x2={yAxisX} y2={height - padding} stroke="#374151" strokeWidth={1.5} />
   317→   317→        {path && <path d={path} fill="none" stroke="#4F46E5" strokeWidth={2.5} strokeLinecap="round" strokeLinejoin="round" />}
   318→   318→        <text x={width / 2} y={height - 5} fontSize={11} fill="#374151" textAnchor="middle" fontWeight={600}>{xLabel}</text>
   319→   319→        <text x={12} y={height / 2} fontSize={11} fill="#374151" textAnchor="middle" fontWeight={600} transform={`rotate(-90, 12, ${height / 2})`}>{yLabel}</text>
   320→   320→      </svg>
   321→   321→    </div>
   322→   322→  );
   323→   323→}

   325→   325→// =====================================================================
   326→   326→// 2. Scatter plot + optional line of best fit (DATA POINTS)
   327→   327→// This directly addresses the user's physics velocity-time request.
   328→   328→// =====================================================================
   329→   329→function ScatterSVG({ spec }: { spec: any }) {
   330→   330→  const title = spec.title ?? "Scatter Plot";
   331→   331→  const xLabel = spec.xLabel ?? "x";
   332→   332→  const yLabel = spec.yLabel ?? "y";
   333→   333→  const points: Array<[number, number]> = Array.isArray(spec.points) ? spec.points : [];
   334→   334→  const showBestFit = spec.lineOfBestFit !== false;
   335→   335→
   336→   336→  // Compute ranges from data (with optional override)
   337→   337→  const allX = points.map((p) => p[0]);
   338→   338→  const allY = points.map((p) => p[1]);
   339→   339→  let xMin = spec.xRange?.[0] ?? Math.min(...allX);
   340→   340→  let xMax = spec.xRange?.[1] ?? Math.max(...allX);
   341→   341→  let yMin = spec.yRange?.[0] ?? Math.min(...allY);
   342→   342→  let yMax = spec.yRange?.[1] ?? Math.max(...allY);
   343→   343→  // Padding
   344→   344→  const xPad = (xMax - xMin) * 0.1 || 1;
   345→   345→  const yPad = (yMax - yMin) * 0.1 || 1;
   346→   346→  if (!spec.xRange) { xMin -= xPad; xMax += xPad; }
   347→   347→  if (!spec.yRange) { yMin -= yPad; yMax += yPad; }
   348→   348→  if (yMin > 0 && spec.yRange?.[0] === undefined) yMin = 0;
   349→   349→  if (xMin > 0 && spec.xRange?.[0] === undefined) xMin = 0;
   350→   350→
   351→   351→  const width = 480;
   352→   352→  const height = 320;
   353→   353→  const padding = 50;
   354→   354→
   355→   355→  const toSvgX = (x: number) => padding + ((x - xMin) / (xMax - xMin)) * (width - 2 * padding);
   356→   356→  const toSvgY = (y: number) =>
   357→   357→    height - padding - ((y - yMin) / (yMax - yMin)) * (height - 2 * padding);
   358→   358→
   359→   359→  // Linear regression for best fit line: y = mx + b
   360→   360→  let bestFitPath: string | null = null;
   361→   361→  let slope = 0;
   362→   362→  let intercept = 0;
   363→   363→  if (showBestFit && points.length >= 2) {
   364→   364→    const n = points.length;
   365→   365→    const sumX = allX.reduce((a, b) => a + b, 0);
   366→   366→    const sumY = allY.reduce((a, b) => a + b, 0);
   367→   367→    const sumXY = points.reduce((s, p) => s + p[0] * p[1], 0);
   368→   368→    const sumXX = allX.reduce((s, x) => s + x * x, 0);
   369→   369→    const denom = n * sumXX - sumX * sumX;
   370→   370→    if (denom !== 0) {
   371→   371→      slope = (n * sumXY - sumX * sumY) / denom;
   372→   372→      intercept = (sumY - slope * sumX) / n;
   373→   373→      const yAtXMin = slope * xMin + intercept;
   374→   374→      const yAtXMax = slope * xMax + intercept;
   375→   375→      bestFitPath = `M ${toSvgX(xMin).toFixed(2)} ${toSvgY(yAtXMin).toFixed(2)} L ${toSvgX(xMax).toFixed(2)} ${toSvgY(yAtXMax).toFixed(2)}`;
   376→   376→    }
   377→   377→  }
   378→   378→
   379→   379→  // Ticks
   380→   380→  const xTicks: ReactElement[] = [];
   381→   381→  const yTicks: ReactElement[] = [];
   382→   382→  const tickCount = 5;
   383→   383→  for (let i = 0; i <= tickCount; i++) {
   384→   384→    const xv = xMin + ((xMax - xMin) * i) / tickCount;
   385→   385→    xTicks.push(
   386→   386→      <g key={`xt-${i}`}>
   387→   387→        <line x1={toSvgX(xv)} y1={toSvgY(yMin) - 4} x2={toSvgX(xv)} y2={toSvgY(yMin) + 4} stroke="#9CA3AF" strokeWidth={1} />
   388→   388→        <text x={toSvgX(xv)} y={toSvgY(yMin) + 18} fontSize={10} fill="#6B7280" textAnchor="middle">
   389→   389→          {Number.isInteger(xv) ? xv : xv.toFixed(1)}
   390→   390→        </text>
   391→   391→      </g>
   392→   392→    );
   393→   393→  }
   394→   394→  for (let i = 0; i <= tickCount; i++) {
   395→   395→    const yv = yMin + ((yMax - yMin) * i) / tickCount;
   396→   396→    yTicks.push(
   397→   397→      <g key={`yt-${i}`}>
   398→   398→        <line x1={toSvgX(xMin) - 4} y1={toSvgY(yv)} x2={toSvgX(xMin) + 4} y2={toSvgY(yv)} stroke="#9CA3AF" strokeWidth={1} />
   399→   399→        <text x={toSvgX(xMin) - 8} y={toSvgY(yv) + 3} fontSize={10} fill="#6B7280" textAnchor="end">
   400→   400→          {Number.isInteger(yv) ? yv : yv.toFixed(0)}
   401→   401→        </text>
   402→   402→      </g>
   403→   403→    );
   404→   404→  }
   405→   405→
   406→   406→  return (
   407→   407→    <div>
   408→   408→      <p className="text-xs font-semibold text-gray-700 mb-1">{title}</p>
   409→   409→      <svg viewBox={`0 0 ${width} ${height}`} className="w-full h-auto bg-gray-50 rounded-lg">
   410→   410→        {/* Grid lines */}
   411→   411→        {xTicks}
   412→   412→        {yTicks}
   413→   413→        {/* Axes */}
   414→   414→        <line x1={toSvgX(xMin)} y1={toSvgY(yMin)} x2={toSvgX(xMax)} y2={toSvgY(yMin)} stroke="#374151" strokeWidth={1.5} />

   536→   536→  const xMin = Math.min(...bins.map((b) => b.start));
   537→   537→  const xMax = Math.max(...bins.map((b) => b.end));

   590→   590→// 5. Pie chart (with labeled slices)
   591→   591→// =====================================================================
   592→   592→function PieChartSVG({ spec }: { spec: any }) {
   593→   593→  const title = spec.title ?? "Pie Chart";
   594→   594→  const slices: Array<{ label: string; value: number; color?: string }> = Array.isArray(spec.slices)
   595→   595→    ? spec.slices
   596→   596→    : [];
   597→   597→  const total = slices.reduce((s, sl) => s + (sl.value ?? 0), 0) || 1;
   598→   598→  const width = 420;
   599→   599→  const height = 320;
   600→   600→  const cx = 160;
   601→   601→  const cy = 160;
   602→   602→  const r = 110;
   603→   603→
   604→   604→  let cumulativeAngle = -Math.PI / 2; // start at 12 o'clock
   605→   605→  const arcs: ReactElement[] = [];
   606→   606→  const legend: ReactElement[] = [];
   607→   607→  slices.forEach((sl, i) => {
   608→   608→    const angle = (sl.value / total) * 2 * Math.PI;
   609→   609→    const x1 = cx + r * Math.cos(cumulativeAngle);
   610→   610→    const y1 = cy + r * Math.sin(cumulativeAngle);
   611→   611→    const x2 = cx + r * Math.cos(cumulativeAngle + angle);
   612→   612→    const y2 = cy + r * Math.sin(cumulativeAngle + angle);
   613→   613→    const largeArc = angle > Math.PI ? 1 : 0;
   614→   614→    const color = sl.color ?? PALETTE[i % PALETTE.length];
   615→   615→    const path = `M ${cx} ${cy} L ${x1.toFixed(2)} ${y1.toFixed(2)} A ${r} ${r} 0 ${largeArc} 1 ${x2.toFixed(2)} ${y2.toFixed(2)} Z`;
   616→   616→    arcs.push(
   617→   617→      <path key={`arc-${i}`} d={path} fill={color} stroke="white" strokeWidth={2} opacity={0.9} />
   618→   618→    );
   619→   619→    // Percentage label at slice center
   620→   620→    const midAngle = cumulativeAngle + angle / 2;
   621→   621→    const labelX = cx + (r * 0.65) * Math.cos(midAngle);
   622→   622→    const labelY = cy + (r * 0.65) * Math.sin(midAngle);
   623→   623→    const pct = ((sl.value / total) * 100).toFixed(1);
   624→   624→    arcs.push(
   625→   625→      <text key={`pct-${i}`} x={labelX} y={labelY + 4} fontSize={11} fill="white" textAnchor="middle" fontWeight={700}>
   626→   626→        {pct}%
   627→   627→      </text>
   628→   628→    );
   629→   629→    // Legend
   630→   630→    legend.push(
   631→   631→      <div key={`leg-${i}`} className="flex items-center gap-1.5 text-[10px]">
   632→   632→        <span style={{ background: color, width: 10, height: 10, borderRadius: 2, display: "inline-block" }} />
   633→   633→        <span className="text-gray-700">{sl.label} ({pct}%)</span>
   634→   634→      </div>
   635→   635→    );
   636→   636→    cumulativeAngle += angle;
   637→   637→  });
   638→   638→
   639→   639→  return (
   640→   640→    <div>
   641→   641→      <p className="text-xs font-semibold text-gray-700 mb-1">{title}</p>
   642→   642→      <div className="flex flex-col sm:flex-row gap-2">
   643→   643→        <svg viewBox={`0 0 ${width} ${height}`} className="w-full sm:w-1/2 h-auto bg-gray-50 rounded-lg">
   644→   644→          {arcs}
   645→   645→          <text x={cx} y={cy + 4} fontSize={11} fill="#374151" textAnchor="middle" fontWeight={700}>Total: {total}</text>
   646→   646→        </svg>
   647→   647→        <div className="flex-1 space-y-1">{legend}</div>
   648→   648→      </div>
   649→   649→    </div>
   650→   650→  );
   651→   651→}

   802→   802→// =====================================================================
   803→   803→// 8. Tree diagram (probability)
   804→   804→// =====================================================================
   805→   805→type TreeNode = {
   806→   806→  label: string;
   807→   807→  probability?: string;
   808→   808→  children?: TreeNode[];
   809→   809→};
   810→   810→
   811→   811→function TreeSVG({ spec }: { spec: any }) {
   812→   812→  const title = spec.title ?? "Tree Diagram";
   813→   813→  const root: TreeNode = spec.root ?? { label: "Start" };
   814→   814→
   815→   815→  const width = 600;
   816→   816→  const height = 380;
   817→   817→  const padding = 30;
   818→   818→
   819→   819→  // Layout: BFS to assign positions
   820→   820→  // depth 0 = root at left
   821→   821→  // each level adds more nodes to the right
   822→   822→  type LayoutNode = {
   823→   823→    node: TreeNode;
   824→   824→    x: number;
   825→   825→    y: number;
   826→   826→    depth: number;
   827→   827→    parent?: { x: number; y: number; probability?: string };
   828→   828→  };
   829→   829→
   830→   830→  const layoutNodes: LayoutNode[] = [];
   831→   831→  let yCounter = 0;
   832→   832→  const maxDepth = 3;
   833→   833→
   834→   834→  function place(node: TreeNode, depth: number, parent?: { x: number; y: number; probability?: string }) {
   835→   835→    if (depth > maxDepth) return;
   836→   836→    const x = padding + depth * 180;
   837→   837→    if (!node.children || node.children.length === 0) {
   838→   838→      const y = padding + yCounter * 36;
   839→   839→      yCounter++;
   840→   840→      layoutNodes.push({ node, x, y, depth, parent });
   841→   841→    } else {
   842→   842→      // first compute children positions to find midpoint
   843→   843→      const childYs: number[] = [];
   844→   844→      for (const child of node.children) {
   845→   845→        const before = yCounter;
   846→   846→        place(child, depth + 1, { x, y: 0, probability: child.probability });
   847→   847→        // after recursive call, find child y — it's the last leaf pushed
   848→   848→        // Take the average of new leaves
   849→   849→        const after = yCounter;
   850→   850→        const midY = (before + after - 1) / 2;
   851→   851→        childYs.push(padding + midY * 36);
   852→   852→      }
   853→   853→      const midY = childYs.reduce((a, b) => a + b, 0) / childYs.length;
   854→   854→      layoutNodes.push({ node, x, y: midY, depth, parent });
   855→   855→    }
   856→   856→  }
   857→   857→
   858→   858→  // Simpler approach: do it iteratively for ≤3 levels
   859→   859→  // Layout children of a node vertically centered around parent.y
   860→   860→  function layout(node: TreeNode, depth: number, parentY: number, parentX: number, probability?: string) {
   861→   861→    if (depth > maxDepth) return;
   862→   862→    const x = padding + depth * 170;
   863→   863→    if (!node.children || node.children.length === 0) {
   864→   864→      layoutNodes.push({ node, x, y: parentY, depth, parent: parentX > 0 ? { x: parentX, y: parentY, probability } : undefined });
   865→   865→      return;
   866→   866→    }
   867→   867→    const childCount = node.children.length;
   868→   868→    const totalHeight = childCount * 50;
   869→   869→    const startY = parentY - totalHeight / 2 + 25;
   870→   870→    const childYs: number[] = [];
   871→   871→    node.children.forEach((child, i) => {
   872→   872→      const cy = startY + i * 50;
   873→   873→      childYs.push(cy);
   874→   874→    });
   875→   875→    const myY = childYs.reduce((a, b) => a + b, 0) / childYs.length;
   876→   876→    layoutNodes.push({ node, x, y: myY, depth, parent: parentX > 0 ? { x: parentX, y: parentY, probability } : undefined });
   877→   877→    node.children.forEach((child, i) => {
   878→   878→      layout(child, depth + 1, childYs[i], x, child.probability);
   879→   879→    });
   880→   880→  }
   881→   881→
   882→   882→  layout(root, 0, height / 2, 0);
   883→   883→
   884→   884→  return (
   885→   885→    <div>
   886→   886→      <p className="text-xs font-semibold text-gray-700 mb-1">{title}</p>
   887→   887→      <svg viewBox={`0 0 ${width} ${height}`} className="w-full h-auto bg-gray-50 rounded-lg">
   888→   888→        {/* Edges */}
   889→   889→        {layoutNodes.map((ln, i) => {
   890→   890→          if (!ln.parent) return null;
   891→   891→          return (
   892→   892→            <g key={`edge-${i}`}>
   893→   893→              <line
   894→   894→                x1={ln.parent.x}
   895→   895→                y1={ln.parent.y}
   896→   896→                x2={ln.x}
   897→   897→                y2={ln.y}
   898→   898→                stroke="#9CA3AF"
   899→   899→                strokeWidth={1.5}
   900→   900→              />
   901→   901→              {ln.parent.probability && (
   902→   902→                <text
   903→   903→                  x={(ln.parent.x + ln.x) / 2}
   904→   904→                  y={(ln.parent.y + ln.y) / 2 - 4}
   905→   905→                  fontSize={10}
   906→   906→                  fill="#6B7280"
   907→   907→                  textAnchor="middle"
   908→   908→                  fontWeight={600}
   909→   909→                >
   910→   910→                  {ln.parent.probability}
   911→   911→                </text>
   912→   912→              )}
   913→   913→            </g>
   914→   914→          );
   915→   915→        })}
   916→   916→        {/* Nodes */}
   917→   917→        {layoutNodes.map((ln, i) => (
   918→   918→          <g key={`node-${i}`}>
   919→   919→            <rect
   920→   920→              x={ln.x - 30}
   921→   921→              y={ln.y - 12}
   922→   922→              width={60}
   923→   923→              height={24}
   924→   924→              rx={12}
   925→   925→              fill={PALETTE[ln.depth % PALETTE.length]}
   926→   926→              opacity={0.9}
   927→   927→            />
   928→   928→            <text x={ln.x} y={ln.y + 4} fontSize={11} fill="white" textAnchor="middle" fontWeight={700}>
   929→   929→              {ln.node.label}
   930→   930→            </text>
   931→   931→          </g>
   932→   932→        ))}
   933→   933→      </svg>
   934→   934→    </div>
   935→   935→  );
   936→   936→}

   941→   941→function NetworkSVG({ spec }: { spec: any }) {
   942→   942→  const title = spec.title ?? "Network Graph";
   943→   943→  const nodes: Array<{ id: string; label: string; color?: string }> = Array.isArray(spec.nodes) ? spec.nodes : [];
   944→   944→  const edges: Array<{ from: string; to: string; label?: string; directed?: boolean }> = Array.isArray(spec.edges) ? spec.edges : [];
   945→   945→
   946→   946→  const width = 480;
   947→   947→  const height = 360;
   948→   948→  const cx = width / 2;
   949→   949→  const cy = height / 2;
   950→   950→  const radius = Math.min(width, height) / 2 - 70;
   951→   951→
   952→   952→  // Detect hub-and-spoke layout
   953→   953→  const nodeDegree = new Map<string, number>();
   954→   954→  for (const e of edges) {
   955→   955→    nodeDegree.set(e.from, (nodeDegree.get(e.from) ?? 0) + 1);
   956→   956→    nodeDegree.set(e.to, (nodeDegree.get(e.to) ?? 0) + 1);
   957→   957→  }
   958→   958→  let hubId: string | null = null;
   959→   959→  if (nodes.length >= 4) {
   960→   960→    const sortedByDegree = [...nodeDegree.entries()].sort((a, b) => b[1] - a[1]);
   961→   961→    if (sortedByDegree.length > 0 && sortedByDegree[0][1] >= 3) {
   962→   962→      const topDegree = sortedByDegree[0][1];
   963→   963→      const secondDegree = sortedByDegree[1]?.[1] ?? 0;
   964→   964→      if (topDegree >= secondDegree * 2) {
   965→   965→        hubId = sortedByDegree[0][0];
   966→   966→      }
   967→   967→    }
   968→   968→  }
   969→   969→
   970→   970→  // Compute INITIAL positions (auto-layout) — stored in state so user can drag
   971→   971→  const computeInitial = useCallback(() => {
   972→   972→    let pos: Array<{ x: number; y: number }>;
   973→   973→    if (hubId) {
   974→   974→      const hubIdx = nodes.findIndex((n) => n.id === hubId);
   975→   975→      const otherCount = nodes.length - 1;
   976→   976→      const angleStep = (2 * Math.PI) / Math.max(otherCount, 1);
   977→   977→      const startAngle = -Math.PI / 2;
   978→   978→      pos = nodes.map((n) => ({ x: cx, y: cy }));
   979→   979→      let otherIdx = 0;
   980→   980→      for (let i = 0; i < nodes.length; i++) {
   981→   981→        if (i === hubIdx) {
   982→   982→          pos[i] = { x: cx, y: cy };
   983→   983→        } else {
   984→   984→          const angle = startAngle + otherIdx * angleStep;
   985→   985→          pos[i] = {
   986→   986→            x: cx + radius * Math.cos(angle),
   987→   987→            y: cy + radius * Math.sin(angle),
   988→   988→          };
   989→   989→          otherIdx++;
   990→   990→        }
   991→   991→      }
   992→   992→    } else {
   993→   993→      pos = nodes.map((_, i) => {
   994→   994→        const angle = (2 * Math.PI * i) / Math.max(nodes.length, 1) - Math.PI / 2;
   995→   995→        return {
   996→   996→          x: cx + radius * Math.cos(angle),
   997→   997→          y: cy + radius * Math.sin(angle),
   998→   998→        };
   999→   999→      });
  1000→  1000→    }
  1001→  1001→    return pos;
  1002→  1002→  }, [hubId, nodes, cx, cy, radius]);
  1003→  1003→
  1004→  1004→  const [positions, setPositions] = useState<Array<{ x: number; y: number }>>(computeInitial);
  1005→  1005→  // Reset positions when spec changes (new conversation)
  1006→  1006→  useEffect(() => {
  1007→  1007→    setPositions(computeInitial());
  1008→  1008→  }, [computeInitial]);
  1009→  1009→
  1010→  1010→  // Drag state
  1011→  1011→  const [dragging, setDragging] = useState<string | null>(null);
  1012→  1012→  const svgRef = useRef<SVGSVGElement | null>(null);
  1013→  1013→
  1014→  1014→  const onNodeMouseDown = (id: string, e: React.MouseEvent) => {
  1015→  1015→    e.preventDefault();
  1016→  1016→    setDragging(id);
  1017→  1017→  };
  1018→  1018→
  1019→  1019→  const onMouseMove = (e: React.MouseEvent) => {
  1020→  1020→    if (!dragging || !svgRef.current) return;
  1021→  1021→    const rect = svgRef.current.getBoundingClientRect();
  1022→  1022→    // Convert mouse coords to SVG viewBox coords
  1023→  1023→    const scale = width / rect.width;
  1024→  1024→    const svgX = (e.clientX - rect.left) * scale;
  1025→  1025→    const svgY = (e.clientY - rect.top) * (height / rect.height);
  1026→  1026→    setPositions((prev) =>
  1027→  1027→      prev.map((p, i) => (nodes[i]?.id === dragging ? { x: svgX, y: svgY } : p))
  1028→  1028→    );
  1029→  1029→  };
  1030→  1030→
  1031→  1031→  const onMouseUp = () => setDragging(null);
  1032→  1032→
  1033→  1033→  return (
  1034→  1034→    <div>
  1035→  1035→      {title && <p className="text-xs font-semibold text-gray-700 mb-1">{title}</p>}
  1036→  1036→      <svg
  1037→  1037→        viewBox={`0 0 ${width} ${height}`}
  1038→  1038→        className="w-full h-auto bg-gray-50 rounded-lg select-none"
  1039→  1039→        ref={svgRef}
  1040→  1040→        onMouseMove={onMouseMove}
  1041→  1041→        onMouseUp={onMouseUp}
  1042→  1042→        onMouseLeave={onMouseUp}
  1043→  1043→        style={{ touchAction: "none" }}
  1044→  1044→      >
  1045→  1045→        {/* Edges */}
  1046→  1046→        {edges.map((edge, i) => {
  1047→  1047→          const fromIdx = nodes.findIndex((n) => n.id === edge.from);
  1048→  1048→          const toIdx = nodes.findIndex((n) => n.id === edge.to);
  1049→  1049→          if (fromIdx < 0 || toIdx < 0) return null;
  1050→  1050→          const from = positions[fromIdx];
  1051→  1051→          const to = positions[toIdx];
  1052→  1052→          const midX = (from.x + to.x) / 2;
  1053→  1053→          const midY = (from.y + to.y) / 2;
  1054→  1054→          return (
  1055→  1055→            <g key={`edge-${i}`}>
  1056→  1056→              <line x1={from.x} y1={from.y} x2={to.x} y2={to.y} stroke="#9CA3AF" strokeWidth={1.5} />
  1057→  1057→              {edge.directed && (
  1058→  1058→                <polygon
  1059→  1059→                  points={arrowPoints(from.x, from.y, to.x, to.y, 8)}
  1060→  1060→                  fill="#9CA3AF"
  1061→  1061→                />
  1062→  1062→              )}
  1063→  1063→              {edge.label && (
  1064→  1064→                <text x={midX} y={midY - 4} fontSize={10} fill="#6B7280" textAnchor="middle" fontWeight={600}>
  1065→  1065→                  {edge.label}
  1066→  1066→                </text>
  1067→  1067→              )}
  1068→  1068→            </g>
  1069→  1069→          );
  1070→  1070→        })}
  1071→  1071→        {/* Nodes (draggable) */}
  1072→  1072→        {nodes.map((n, i) => {

  1074→  1074→          const isHub = n.id === hubId;
  1075→  1075→          const isDragging = dragging === n.id;
  1076→  1076→          const color = n.color ?? PALETTE[i % PALETTE.length];
  1077→  1077→          const fontSize = isHub ? 12 : 11;
  1078→  1078→          const padding = isHub ? 12 : 8;
  1079→  1079→          const rectHeight = isHub ? 36 : 32;
  1080→  1080→          const labelWidth = Math.max(isHub ? 100 : 60, n.label.length * 7 + padding);
  1081→  1081→          return (
  1082→  1082→            <g
  1083→  1083→              key={n.id}
  1084→  1084→              onMouseDown={(e) => onNodeMouseDown(n.id, e)}
  1085→  1085→              style={{ cursor: isDragging ? "grabbing" : "grab" }}
  1086→  1086→            >
  1087→  1087→              {isHub && (
  1088→  1088→                // Glow / highlight ring around the hub
  1089→  1089→                <rect
  1090→  1090→                  x={pos.x - labelWidth / 2 - 3}
  1091→  1091→                  y={pos.y - rectHeight / 2 - 3}
  1092→  1092→                  width={labelWidth + 6}
  1093→  1093→                  height={rectHeight + 6}
  1094→  1094→                  rx={(rectHeight + 6) / 2}
  1095→  1095→                  fill="none"
  1096→  1096→                  stroke={color}
  1097→  1097→                  strokeWidth={1.5}
  1098→  1098→                  strokeOpacity={0.4}
  1099→  1099→                />
  1100→  1100→              )}
  1101→  1101→              <rect
  1102→  1102→                x={pos.x - labelWidth / 2}
  1103→  1103→                y={pos.y - rectHeight / 2}
  1104→  1104→                width={labelWidth}
  1105→  1105→                height={rectHeight}
  1106→  1106→                rx={rectHeight / 2}
  1107→  1107→                fill={color}

  1109→  1109→              />
  1110→  1110→              <text

  1114→  1114→                fill="white"

  1132→  1132→function arrowPoints(fromX: number, fromY: number, toX: number, toY: number, size: number): string {
  1133→  1133→  // shorten endpoint to node radius
  1134→  1134→  const dx = toX - fromX;
  1135→  1135→  const dy = toY - fromY;
  1136→  1136→  const len = Math.sqrt(dx * dx + dy * dy) || 1;
  1137→  1137→  const nx = dx / len;
  1138→  1138→  const ny = dy / len;
  1139→  1139→  const tipX = toX - nx * 18;
  1140→  1140→  const tipY = toY - ny * 18;
  1141→  1141→  const baseX = tipX - nx * size;
  1142→  1142→  const baseY = tipY - ny * size;
  1143→  1143→  const perpX = -ny * size * 0.5;
  1144→  1144→  const perpY = nx * size * 0.5;
  1145→  1145→  return `${tipX},${tipY} ${baseX + perpX},${baseY + perpY} ${baseX - perpX},${baseY - perpY}`;
  1146→  1146→}

  1214→  1214→        <text x={width / 2} y={height - 8} fontSize={11} fill="#374151" textAnchor="middle" fontWeight={600}>{xLabel}</text>
  1215→  1215→        <text x={12} y={height / 2} fontSize={11} fill="#374151" textAnchor="middle" fontWeight={600} transform={`rotate(-90, 12, ${height / 2})`}>{yLabel}</text>
  1216→  1216→      </svg>

  1222→  1222→// 11. Polygon (2D geometric figure with labeled vertices and sides)

  1224→  1224→function PolygonSVG({ spec }: { spec: any }) {
  1225→  1225→  const title = spec.title ?? "Polygon";
  1226→  1226→  const vertices: Array<[number, number]> = Array.isArray(spec.vertices) ? spec.vertices : [];
  1227→  1227→  const labels: string[] = Array.isArray(spec.labels) ? spec.labels : [];
  1228→  1228→  const showAngles = spec.showAngles !== false;
  1229→  1229→  const showSides = spec.showSides !== false;
  1230→  1230→
  1231→  1231→  // Compute bounds
  1232→  1232→  const allX = vertices.map((v) => v[0]);
  1233→  1233→  const allY = vertices.map((v) => v[1]);
  1234→  1234→  let xMin = Math.min(...allX);
  1235→  1235→  let xMax = Math.max(...allX);
  1236→  1236→  let yMin = Math.min(...allY);
  1237→  1237→  let yMax = Math.max(...allY);
  1238→  1238→  const xPad = (xMax - xMin) * 0.2 || 1;
  1239→  1239→  const yPad = (yMax - yMin) * 0.2 || 1;
  1240→  1240→  xMin -= xPad; xMax += xPad; yMin -= yPad; yMax += yPad;
  1241→  1241→
  1242→  1242→  const width = 380;
  1243→  1243→  const height = 320;
  1244→  1244→  const padding = 30;
  1245→  1245→
  1246→  1246→  const toSvgX = (x: number) => padding + ((x - xMin) / (xMax - xMin)) * (width - 2 * padding);
  1247→  1247→  const toSvgY = (y: number) =>
  1248→  1248→    height - padding - ((y - yMin) / (yMax - yMin)) * (height - 2 * padding);
  1249→  1249→
  1250→  1250→  const path = vertices.map((v, i) => `${i === 0 ? "M" : "L"} ${toSvgX(v[0]).toFixed(2)} ${toSvgY(v[1]).toFixed(2)}`).join(" ") + " Z";
  1251→  1251→
  1252→  1252→  // Side lengths
  1253→  1253→  const sides: Array<{ length: number; midX: number; midY: number }> = [];
  1254→  1254→  for (let i = 0; i < vertices.length; i++) {
  1255→  1255→    const a = vertices[i];
  1256→  1256→    const b = vertices[(i + 1) % vertices.length];
  1257→  1257→    const len = Math.sqrt((b[0] - a[0]) ** 2 + (b[1] - a[1]) ** 2);
  1258→  1258→    sides.push({
  1259→  1259→      length: len,
  1260→  1260→      midX: toSvgX((a[0] + b[0]) / 2),
  1261→  1261→      midY: toSvgY((a[1] + b[1]) / 2),
  1262→  1262→    });
  1263→  1263→  }
  1264→  1264→
  1265→  1265→  return (
  1266→  1266→    <div>
  1267→  1267→      <p className="text-xs font-semibold text-gray-700 mb-1">{title}</p>
  1268→  1268→      <svg viewBox={`0 0 ${width} ${height}`} className="w-full h-auto bg-gray-50 rounded-lg">
  1269→  1269→        {/* Polygon */}
  1270→  1270→        <path d={path} fill="#4F46E5" fillOpacity={0.15} stroke="#4F46E5" strokeWidth={2.5} />
  1271→  1271→        {/* Vertices + labels */}
  1272→  1272→        {vertices.map((v, i) => {
  1273→  1273→          const label = labels[i] ?? String.fromCharCode(65 + i);
  1274→  1274→          // Offset label outward (from centroid)
  1275→  1275→          const cx = vertices.reduce((s, p) => s + p[0], 0) / vertices.length;
  1276→  1276→          const cy = vertices.reduce((s, p) => s + p[1], 0) / vertices.length;
  1277→  1277→          const dx = v[0] - cx;
  1278→  1278→          const dy = v[1] - cy;
  1279→  1279→          const len = Math.sqrt(dx * dx + dy * dy) || 1;
  1280→  1280→          const labelOffset = 16;
  1281→  1281→          const labelX = toSvgX(v[0]) + (dx / len) * labelOffset;
  1282→  1282→          const labelY = toSvgY(v[1]) + (dy / len) * labelOffset;
  1283→  1283→          return (
  1284→  1284→            <g key={`v-${i}`}>
  1285→  1285→              <circle cx={toSvgX(v[0])} cy={toSvgY(v[1])} r={3.5} fill="#4F46E5" />
  1286→  1286→              <text x={labelX} y={labelY + 4} fontSize={13} fill="#374151" textAnchor="middle" fontWeight={700}>
  1287→  1287→                {label}
  1288→  1288→              </text>
  1289→  1289→            </g>
  1290→  1290→          );
  1291→  1291→        })}
  1292→  1292→        {/* Side length labels */}
  1293→  1293→        {showSides && sides.map((s, i) => (
  1294→  1294→          <text key={`s-${i}`} x={s.midX} y={s.midY - 4} fontSize={10} fill="#10B981" textAnchor="middle" fontWeight={600}>
  1295→  1295→            {s.length.toFixed(2)}
  1296→  1296→          </text>
  1297→  1297→        ))}
  1298→  1298→      </svg>
  1299→  1299→    </div>
  1300→  1300→  );
  1301→  1301→}

  1401→  1401→// 13. Slope field (direction field) for differential equations
  1402→  1402→//     Draws small tick marks showing the slope of y' = f(x, y) at grid points.
  1403→  1403→// =====================================================================
  1404→  1404→function SlopeFieldSVG({ spec }: { spec: any }) {
  1405→  1405→  const title = spec.title ?? "Slope Field";
  1406→  1406→  const expr = spec.expr ?? "x"; // dy/dx = expr (function of x and y)
  1407→  1407→  const xRange: [number, number] = spec.xRange ?? [-5, 5];
  1408→  1408→  const yRange: [number, number] = spec.yRange ?? [-5, 5];
  1409→  1409→  const gridSize: number = spec.gridSize ?? 10; // grid points per axis
  1410→  1410→  const xLabel = spec.xLabel ?? "x";
  1411→  1411→  const yLabel = spec.yLabel ?? "y";
  1412→  1412→  const width = 480;
  1413→  1413→  const height = 360;
  1414→  1414→  const padding = 40;
  1415→  1415→
  1416→  1416→  // Safe expression evaluator — supports Math.* + x + y
  1417→  1417→  const evaluate = (x: number, y: number): number | null => {
  1418→  1418→    try {
  1419→  1419→      let safeExpr = expr
  1420→  1420→        .replace(/\^/g, "**")
  1421→  1421→        .replace(/\bpi\b/gi, "Math.PI")
  1422→  1422→        .replace(/\be\b/g, "Math.E")
  1423→  1423→        .replace(/\bsin\(/g, "Math.sin(")
  1424→  1424→        .replace(/\bcos\(/g, "Math.cos(")
  1425→  1425→        .replace(/\btan\(/g, "Math.tan(")
  1426→  1426→        .replace(/\bsqrt\(/g, "Math.sqrt(")
  1427→  1427→        .replace(/\blog\(/g, "Math.log(")
  1428→  1428→        .replace(/\bexp\(/g, "Math.exp(")
  1429→  1429→        .replace(/\babs\(/g, "Math.abs(")
  1430→  1430→        .replace(/\bx\b/g, String(x))
  1431→  1431→        .replace(/\by\b/g, String(y));
  1432→  1432→      // eslint-disable-next-line no-new-func
  1433→  1433→      const fn = new Function("Math", `"use strict"; return (${safeExpr});`);
  1434→  1434→      const result = fn(Math);
  1435→  1435→      return typeof result === "number" && isFinite(result) ? result : null;
  1436→  1436→    } catch {
  1437→  1437→      return null;
  1438→  1438→    }
  1439→  1439→  };
  1440→  1440→
  1441→  1441→  const toSvgX = (x: number) => padding + ((x - xRange[0]) / (xRange[1] - xRange[0])) * (width - 2 * padding);
  1442→  1442→  const toSvgY = (y: number) =>
  1443→  1443→    height - padding - ((y - yRange[0]) / (yRange[1] - yRange[0])) * (height - 2 * padding);
  1444→  1444→
  1445→  1445→  // Generate slope tick marks
  1446→  1446→  const ticks: ReactElement[] = [];
  1447→  1447→  const tickLength = 8;
  1448→  1448→  for (let i = 0; i <= gridSize; i++) {
  1449→  1449→    for (let j = 0; j <= gridSize; j++) {
  1450→  1450→      const x = xRange[0] + ((xRange[1] - xRange[0]) * i) / gridSize;
  1451→  1451→      const y = yRange[0] + ((yRange[1] - yRange[0]) * j) / gridSize;
  1452→  1452→      const slope = evaluate(x, y);
  1453→  1453→      if (slope === null) continue;
  1454→  1454→      // Normalize direction
  1455→  1455→      const dx = 1 / Math.sqrt(1 + slope * slope);
  1456→  1456→      const dy = slope / Math.sqrt(1 + slope * slope);
  1457→  1457→      const cx = toSvgX(x);
  1458→  1458→      const cy = toSvgY(y);
  1459→  1459→      const x1 = cx - (tickLength / 2) * dx;
  1460→  1460→      const y1 = cy + (tickLength / 2) * dy; // SVG y inverted
  1461→  1461→      const x2 = cx + (tickLength / 2) * dx;
  1462→  1462→      const y2 = cy - (tickLength / 2) * dy;
  1463→  1463→      ticks.push(
  1464→  1464→        <line
  1465→  1465→          key={`tick-${i}-${j}`}
  1466→  1466→          x1={x1.toFixed(2)}
  1467→  1467→          y1={y1.toFixed(2)}
  1468→  1468→          x2={x2.toFixed(2)}
  1469→  1469→          y2={y2.toFixed(2)}
  1470→  1470→          stroke="#4F46E5"
  1471→  1471→          strokeWidth={1.5}
  1472→  1472→          opacity={0.7}
  1473→  1473→        />
  1474→  1474→      );
  1475→  1475→    }
  1476→  1476→  }
  1477→  1477→
  1478→  1478→  // Axes
  1479→  1479→  const xAxisY = yRange[0] <= 0 && yRange[1] >= 0 ? toSvgY(0) : height - padding;
  1480→  1480→  const yAxisX = xRange[0] <= 0 && xRange[1] >= 0 ? toSvgX(0) : padding;
  1481→  1481→
  1482→  1482→  return (
  1483→  1483→    <div>
  1484→  1484→      <p className="text-xs font-semibold text-gray-700 mb-1">{title}</p>
  1485→  1485→      <p className="text-[10px] text-gray-500 mb-1">dy/dx = <span className="font-mono">{expr}</span></p>
  1486→  1486→      <svg viewBox={`0 0 ${width} ${height}`} className="w-full h-auto bg-gray-50 rounded-lg">
  1487→  1487→        {ticks}
  1488→  1488→        <line x1={padding} y1={xAxisY} x2={width - padding} y2={xAxisY} stroke="#374151" strokeWidth={1.5} />
  1489→  1489→        <line x1={yAxisX} y1={padding} x2={yAxisX} y2={height - padding} stroke="#374151" strokeWidth={1.5} />
  1490→  1490→        <text x={width / 2} y={height - 8} fontSize={11} fill="#374151" textAnchor="middle" fontWeight={600}>{xLabel}</text>
  1491→  1491→        <text x={12} y={height / 2} fontSize={11} fill="#374151" textAnchor="middle" fontWeight={600} transform={`rotate(-90, 12, ${height / 2})`}>{yLabel}</text>
  1492→  1492→      </svg>
  1493→  1493→    </div>
  1494→  1494→  );
  1495→  1495→}

  1498→  1498→// 14. Stem-and-leaf plot (statistical data display)
  1499→  1499→// =====================================================================
  1500→  1500→function StemLeafSVG({ spec }: { spec: any }) {
  1501→  1501→  const title = spec.title ?? "Stem-and-Leaf Plot";
  1502→  1502→  const data: number[] = Array.isArray(spec.data) ? spec.data : [];
  1503→  1503→  const stemUnit: number = spec.stemUnit ?? 10; // e.g. 10 means stem = tens digit
  1504→  1504→  const leafUnit: number = spec.leafUnit ?? 1; // e.g. 1 means leaf = ones digit
  1505→  1505→
  1506→  1506→  if (data.length === 0) {
  1507→  1507→    return <p className="text-xs text-gray-500">No data provided for stem-and-leaf plot.</p>;
  1508→  1508→  }
  1509→  1509→
  1510→  1510→  // Group by stem
  1511→  1511→  const groups = new Map<number, number[]>();
  1512→  1512→  for (const v of data) {
  1513→  1513→    const stem = Math.floor(v / stemUnit);
  1514→  1514→    const leaf = Math.floor((v - stem * stemUnit) / leafUnit);
  1515→  1515→    if (!groups.has(stem)) groups.set(stem, []);
  1516→  1516→    groups.get(stem)!.push(leaf);
  1517→  1517→  }
  1518→  1518→  // Sort stems
  1519→  1519→  const sortedStems = [...groups.keys()].sort((a, b) => a - b);
  1520→  1520→
  1521→  1521→  return (
  1522→  1522→    <div>
  1523→  1523→      <p className="text-xs font-semibold text-gray-700 mb-1">{title}</p>
  1524→  1524→      <p className="text-[10px] text-gray-500 mb-2">Stem = {stemUnit}s, Leaf = {leafUnit}s</p>
  1525→  1525→      <div className="bg-white border border-gray-200 rounded-lg p-3 font-mono text-xs">
  1526→  1526→        <table className="w-full">
  1527→  1527→          <thead>
  1528→  1528→            <tr className="border-b border-gray-200 text-gray-600">
  1529→  1529→              <th className="text-left py-1 px-2">Stem</th>
  1530→  1530→              <th className="text-left py-1 px-2">Leaves</th>
  1531→  1531→              <th className="text-right py-1 px-2 text-gray-400">Count</th>
  1532→  1532→            </tr>
  1533→  1533→          </thead>
  1534→  1534→          <tbody>
  1535→  1535→            {sortedStems.map((stem) => {
  1536→  1536→              const leaves = (groups.get(stem) ?? []).sort((a, b) => a - b);
  1537→  1537→              return (
  1538→  1538→                <tr key={stem} className="border-b border-gray-50">
  1539→  1539→                  <td className="py-1 px-2 font-bold text-indigo-600">{stem} |</td>
  1540→  1540→                  <td className="py-1 px-2 tracking-wide text-gray-800">{leaves.join(" ")}</td>
  1541→  1541→                  <td className="py-1 px-2 text-right text-gray-400">{leaves.length}</td>
  1542→  1542→                </tr>
  1543→  1543→              );
  1544→  1544→            })}
  1545→  1545→          </tbody>
  1546→  1546→        </table>
  1547→  1547→      </div>

  1635→  1635→function FreeformSVG({ spec }: { spec: any }) {

  1693→  1693→  sanitized = sanitized.replace(/href\s*=\s*'https?:\/\/[^']*'/gi, "href='#'");

  1862→  1862→

  1864→  1864→  const eval2 = (expr: string, x: number, y: number): number | null => {

  1867→  1867→        .replace(/\^/g, "**")
