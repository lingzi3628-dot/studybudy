   385→  // =====================================================================
   386→  // Continuous voice conversation mode (like ChatGPT voice mode)
   387→  // =====================================================================
   388→  // When voiceMode is ON:
   389→  //   1. We auto-start listening for the user's question
   390→  //   2. When the user stops, we transcribe and auto-send
   391→  //   3. When the AI replies, we speak it back via browser TTS

   394→
   395→  // Start the voice mode cycle — listen for user's question
   396→  const startVoiceListening = useCallback(() => {
   397→    if (!isBrowserASRSupported()) {
   398→      setError("Voice mode needs Chrome, Edge, or Safari (Firefox doesn't support speech recognition).");
   399→      setVoiceMode(false);

   402→    }

   404→    try {
   405→      // Stop any existing listener
   406→      if (voiceListenerRef.current) {
   407→        voiceListenerRef.current.stop();

   409→      }

   411→        lang: "en-US",
   412→        continuous: false,
   413→        interimResults: false,
   414→        maxDurationSec: 30,
   415→      });
   416→      voiceListenerRef.current = listener;
   417→
   418→      listener.onResult((r) => {
   419→        if (r.isFinal && r.text.trim()) {
   420→          // Stop listener, then send the text via the regular send path
   421→          // which will trigger the AI reply → onReplyFinished → startSpeaking cycle
   422→          setVoiceModeState("speaking");
   423→          setInput(r.text);
   424→          send(r.text);
   425→        }
   426→      });

   428→        const err = e?.error;

   430→          setError("Microphone access denied. Allow mic permission for voice mode.");
   431→          setVoiceMode(false);
   432→          voiceModeRef.current = false;
   433→        } else if (err === "no-speech") {
   434→          // User didn't speak — just restart listening if voiceMode is still on
   435→          if (voiceModeRef.current) {
   436→            setTimeout(() => startVoiceListening(), 300);
   437→          }
   438→        } else if (err === "aborted" || err === "interrupted") {
   439→          // Normal events in voice mode — ASR gets aborted when user stops,
   440→          // TTS gets interrupted when user starts speaking. Don't show errors.
   441→          // Just restart listening if voice mode is still on.
   442→          if (voiceModeRef.current) {
   443→            setTimeout(() => startVoiceListening(), 200);
   444→          }
   445→        } else {
   446→          console.warn("[voice mode] ASR error:", err ?? "unknown");
   447→          // Don't show error banner for minor voice mode issues — just restart
   448→          if (voiceModeRef.current) {
   449→            setTimeout(() => startVoiceListening(), 500);
   450→          }
   451→        }
   452→      });
   453→      listener.onEnd(() => {
   454→        voiceListenerRef.current = null;
   455→        // If voice mode is still on but we haven't transitioned to speaking,
   456→        // restart listening (handles the "no-speech" silent restart path)
   457→        if (voiceModeRef.current && voiceModeState !== "speaking") {
   458→          setTimeout(() => startVoiceListening(), 200);
   459→        }
   460→      });
   461→    } catch (e: any) {
   462→      console.error("[voice mode] startListening failed:", e?.message);
   463→      setVoiceMode(false);
   464→      voiceModeRef.current = false;
   465→    }
   466→  }, [send, voiceModeState]);
   467→
   468→  // Speak the AI reply, then start listening again
   469→  const speakReplyAndContinue = useCallback((text: string) => {
   470→    if (!isBrowserTTSSupported()) {
   471→      console.warn("[voice mode] TTS not supported — skipping speak, going back to listening");
   472→      setTimeout(() => startVoiceListening(), 300);
   473→      return;
   474→    }
   475→    setVoiceModeState("speaking");
   476→    const plainText = text
   477→      .replace(/```[\s\S]*?```/g, " [code block] ")
   478→      .replace(/\$\$[^$]+\$\$/g, " math equation ")
   479→      .replace(/\$([^$]+)\$/g, " $1 ")
   480→      .replace(/[*_`#>|]/g, "")
   481→      .replace(/\[([^\]]+)\]\([^)]+\)/g, "$1")
   482→      .replace(/\s+/g, " ")
   483→      .trim();
   484→    if (!plainText) {
   485→      // Empty reply — go straight back to listening
   486→      setTimeout(() => startVoiceListening(), 200);
   487→      return;
   488→    }
   489→    try {
   490→      const { promise } = browserSpeak(plainText, { rate: 1.0, lang: "en-US" });
   491→      promise.then(() => {
   492→        // TTS finished — start listening again if voice mode is still on
   493→        if (voiceModeRef.current) {
   494→          setTimeout(() => startVoiceListening(), 300);
   495→        }
   496→      }).catch((e: any) => {
   497→        // "interrupted" is normal — happens when TTS is cancelled (e.g. user taps stop)
   498→        if (e?.message !== "interrupted" && e?.message !== "aborted") {
   499→          console.warn("[voice mode] TTS error:", e?.message);
   500→        }
   501→        // Go back to listening even on TTS error
   502→        if (voiceModeRef.current) {
   503→          setTimeout(() => startVoiceListening(), 300);
   504→        }
   505→      });
   506→    } catch (e: any) {
   507→      console.warn("[voice mode] browserSpeak failed:", e?.message);
   508→      if (voiceModeRef.current) {
   509→        setTimeout(() => startVoiceListening(), 300);
   510→      }
   511→    }
   512→  }, [startVoiceListening]);
   513→
   514→  // Watch for new AI replies while in voice mode → speak them
   515→  useEffect(() => {
   516→    if (!voiceMode) return;
   517→    if (busy) return; // wait for the AI to finish
   518→    if (messages.length === 0) return;
   519→    const lastMsg = messages[messages.length - 1];
   520→    if (lastMsg.role !== "assistant") return;
   521→    // Only speak if we haven't already spoken this message
   522→    // (use a data attribute on the message via a ref check)
   523→    // For simplicity, we use a ref to track the last spoken message id
   524→    if (lastSpokenRef.current === lastMsg.id) return;
   525→    lastSpokenRef.current = lastMsg.id;
   526→    speakReplyAndContinue(lastMsg.content);
   527→  }, [messages, voiceMode, busy, speakReplyAndContinue]);
   528→
   529→  const lastSpokenRef = useRef<string | null>(null);
   530→
   531→  const toggleVoiceMode = () => {
   532→    if (voiceMode) {
   533→      // Turn off — stop listening + stop speaking
   534→      if (voiceListenerRef.current) {
   535→        voiceListenerRef.current.stop();
   536→        voiceListenerRef.current = null;
   537→      }
   538→      stopBrowserSpeech();
   539→      setVoiceMode(false);
   540→      voiceModeRef.current = false;
   541→      setVoiceModeState("idle");
   542→    } else {
   543→      // Turn on — start listening

  1567→function MessageBubble({
  1568→  msg,
  1569→  onCopy,
  1570→  onRetry,
  1571→  copied,
  1572→  onAttachmentChange,
  1573→}: {
  1574→  msg: ChatMsg;
  1575→  onCopy: () => void;
  1576→  onRetry?: () => void;
  1577→  copied: boolean;
  1578→  onAttachmentChange?: (attIdx: number, newCaption: string) => void;
  1579→}) {
  1580→  const isUser = msg.role === "user";
  1581→  const [speaking, setSpeaking] = useState(false);
  1582→
  1583→  // Clean up browser speech when component unmounts
  1584→  useEffect(() => {
  1585→    return () => {
  1586→      if (speaking) stopBrowserSpeech();
  1587→    };
  1588→  }, [speaking]);
  1589→
  1590→  const speak = async () => {
  1591→    // If already speaking, stop
  1592→    if (speaking) {
  1593→      stopBrowserSpeech();
  1594→      setSpeaking(false);
  1595→      return;
  1596→    }
  1597→
  1598→    // Strip markdown + LaTeX + attachments for TTS (we only speak the prose)
  1599→    const plainText = msg.content
  1600→      .replace(/```[\s\S]*?```/g, " [code block] ")
  1601→      .replace(/\$\$[^$]+\$\$/g, " math equation ")
  1602→      .replace(/\$([^$]+)\$/g, " $1 ")
  1603→      .replace(/[*_`#>|]/g, "")
  1604→      .replace(/\[([^\]]+)\]\([^)]+\)/g, "$1")
  1605→      .replace(/\s+/g, " ")
  1606→      .trim();

  1613→    if (isBrowserTTSSupported()) {
  1614→      try {
  1615→        const { promise } = browserSpeak(plainText, { rate: 1.0, lang: "en-US" });
  1616→        await promise;
  1617→        setSpeaking(false);
  1618→        return;
  1619→      } catch (e: any) {
  1620→        console.warn("[tutor] browser TTS failed, falling back to server:", e?.message);
  1621→        setSpeaking(false);
  1622→        // Fall through to server fallback below
  1623→      }
  1624→    }
  1625→
  1626→    // FALLBACK: Server-side TTS via z-ai-web-dev-sdk
  1627→    // (only used if browser doesn't support speechSynthesis — very rare)
  1628→    try {
  1629→      const chunks: string[] = [];
  1630→      let remaining = plainText;
  1631→      while (remaining.length > 1000) {
  1632→        const cut = remaining.lastIndexOf(".", 1000);
  1633→        chunks.push(remaining.slice(0, cut > 0 ? cut + 1 : 1000));
  1634→        remaining = remaining.slice(cut > 0 ? cut + 1 : 1000);
  1635→      }
  1636→      chunks.push(remaining);
  1637→
  1638→      const firstChunk = chunks[0];
  1639→      const r = await fetch("/api/tutor/tts", {
  1640→        method: "POST",
  1641→        headers: { "Content-Type": "application/json" },
  1642→        body: JSON.stringify({ text: firstChunk, voice: "tongtong", speed: 1.0 }),
  1643→      });
  1644→      if (!r.ok) {
  1645→        const d = await r.json().catch(() => ({}));
  1646→        throw new Error(d.error ?? "TTS failed");
  1647→      }
  1648→      const blob = await r.blob();
  1649→      const url = URL.createObjectURL(blob);
  1650→      const audio = new Audio(url);
  1651→      audio.onended = () => {
  1652→        setSpeaking(false);
  1653→        URL.revokeObjectURL(url);
  1654→      };
  1655→      audio.onerror = () => {
  1656→        setSpeaking(false);
  1657→        URL.revokeObjectURL(url);
  1658→      };
  1659→      await audio.play();
  1660→    } catch (e: any) {
  1661→      console.error("[tutor] TTS error:", e?.message);
  1662→      // Can't setError here — MessageBubble is a child component.
  1663→      // The browser TTS path is the primary; this only runs if browser
  1664→      // doesn't support speechSynthesis (rare) AND server is also broken.
  1665→      alert(e?.message ?? "Voice playback failed");
  1666→      setSpeaking(false);
  1667→    }
  1668→  };
  1669→
  1670→  return (
  1671→    <div className={`flex ${isUser ? "justify-end" : "justify-start"} group`}>

  1682→                ? "bg-indigo-600 text-white rounded-br-sm"

  1848→// Markdown Renderer — handles code blocks, lists, bold/italic, links

  1850→function MarkdownContent({ content, isUser }: { content: string; isUser: boolean }) {
  1851→  // Split content into blocks: code blocks vs. inline content
  1852→  const blocks: Array<{ type: "code" | "text"; lang?: string; content: string }> = [];
  1853→  const codeBlockRegex = /```(\w+)?\n?([\s\S]*?)```/g;
  1854→  let lastIndex = 0;
  1855→  let match: RegExpExecArray | null;
  1856→
  1857→  while ((match = codeBlockRegex.exec(content)) !== null) {
  1858→    if (match.index > lastIndex) {
  1859→      blocks.push({ type: "text", content: content.slice(lastIndex, match.index) });
  1860→    }
  1861→    blocks.push({
  1862→      type: "code",
  1863→      lang: match[1] || "text",
  1864→      content: match[2].trim(),
  1865→    });
  1866→    lastIndex = match.index + match[0].length;
  1867→  }
  1868→  if (lastIndex < content.length) {
  1869→    blocks.push({ type: "text", content: content.slice(lastIndex) });
  1870→  }
  1871→  // If no blocks were created (no code blocks), use the entire content as text
  1872→  if (blocks.length === 0) {
  1873→    blocks.push({ type: "text", content });
  1874→  }
  1875→
  1876→  return (
  1877→    <div className="space-y-2">
  1878→      {blocks.map((block, i) => {
  1879→        if (block.type === "code") {
  1880→          // Skip mathgraph/conceptmap code blocks — they are rendered as attachments
  1881→          if (block.lang === "mathgraph" || block.lang === "conceptmap" || block.lang === "examgen") return null;
  1882→          // Also skip JSON / text blocks that look like graph specs (since the
  1883→          // server has parsed them into attachments already). Check if the
  1884→          // block content starts with `{"type": "..."` where type is one of
  1885→          // our known graph types.
  1886→          const graphTypeMatch = block.content.match(/^\{\s*"type"\s*:\s*"(\w+)"/);
  1887→          const KNOWN_GRAPH_TYPES = new Set([
  1888→            "function", "scatter", "bar", "histogram", "pie", "venn",
  1889→            "numberline", "tree", "network", "vector", "polygon", "boxplot",
  1890→            "slopefield", "stemleaf", "frequency_polygon", "freeform",
  1891→            "argand", "contour", "vectorfield", "tessellation", "knot",
  1892→            "pictogram", "tally", "carroll", "ogive", "unitcircle",
  1893→            "transform", "axes3d", "twoway", "erdiagram", "csv", "steps",
  1894→          ]);
  1895→          if (
  1896→            (block.lang === "json" || block.lang === "text" || block.lang === "") &&
  1897→            graphTypeMatch && KNOWN_GRAPH_TYPES.has(graphTypeMatch[1])
  1898→          ) {
  1899→            return null;
  1900→          }
  1901→          return <CodeBlock key={i} code={block.content} lang={block.lang} />;
