    69→// CSVDownloadButton — small button that downloads rows as a CSV file
    70→// (opens in Excel, Google Sheets, LibreOffice)
    71→function CSVDownloadButton({
    72→  headers,
    73→  rows,
    74→  downloadName = "table.csv",
    75→}: {
    76→  headers: string[];
    77→  rows: (string | number)[][];
    78→  downloadName?: string;
    79→}) {
    80→  const escapeCSV = (val: string | number) => {
    81→    const s = String(val ?? "");
    82→    if (/[",\n]/.test(s)) return `"${s.replace(/"/g, '""')}"`;
    83→    return s;
    84→  };
    85→  const csvText = [headers, ...rows].map((r) => r.map(escapeCSV).join(",")).join("\n");
    86→
    87→  const download = () => {
    88→    const blob = new Blob([csvText], { type: "text/csv;charset=utf-8" });
    89→    const url = URL.createObjectURL(blob);
    90→    const a = document.createElement("a");
    91→    a.href = url;
    92→    a.download = downloadName.endsWith(".csv") ? downloadName : `${downloadName}.csv`;
    93→    document.body.appendChild(a);
    94→    a.click();
    95→    document.body.removeChild(a);
    96→    URL.revokeObjectURL(url);
    97→  };
    98→
    99→  return (
   100→    <button
   101→      onClick={download}
   102→      className="text-[10px] text-gray-500 hover:text-emerald-700 flex items-center gap-0.5"
   103→      title="Download as CSV (opens in Excel)"
   104→    >
   105→      📄 CSV
   106→    </button>
   107→  );
   108→}

   111→// GraphRenderers — Phase 28+
   112→//
   113→// A modular dispatcher that renders many different kinds of graphs as
   114→// inline SVG, based on a JSON spec produced by the AI Tutor.
   115→//
   116→// Supported types:
   117→//   1. function       — y = f(x) line plot (math)
   118→//   2. scatter        — data points + optional line of best fit (physics/data)
   119→//   3. bar            — vertical bar chart with categories
   120→//   4. histogram      — like bar but no gaps, for frequency distributions
   121→//   5. pie            — pie chart with labeled slices
   122→//   6. venn           — 2 or 3-set Venn diagram (probability/sets)
   123→//   7. numberline     — number line with shaded ranges (inequalities)
   124→//   8. tree           — probability tree diagram
   125→//   9. network        — node/edge graph (graph theory, social networks)
   126→//  10. vector         — vector arrows on a coordinate plane
   127→//  11. polygon        — 2D geometric figure with labeled vertices/angles
   128→//  12. boxplot        — box-and-whisker plot (statistics)
   129→//
   130→// Each renderer takes a `spec: any` prop (the parsed JSON from the AI).
   131→// All are pure SVG so they work in any modern browser, no external deps.
   132→// =====================================================================

   225→// =====================================================================
   226→// 1. Function plot (y = f(x))
   227→// =====================================================================
   228→function FunctionSVG({ spec }: { spec: any }) {
   229→  const expr = spec.expr || "x^2";
   230→  const xRange: [number, number] = spec.xRange ?? [-5, 5];
   231→  const yRange: [number, number] = spec.yRange ?? [-25, 25];
   232→  const title = spec.title || `y = ${expr}`;
   233→  const xLabel = spec.xLabel ?? "x";
   234→  const yLabel = spec.yLabel ?? "y";
   235→  const width = 480;
   236→  const height = 320;
   237→  const padding = 40;
   238→
   239→  const evaluate = (x: number): number | null => {
   240→    try {
   241→      let safeExpr = expr
   242→        .replace(/\^/g, "**")
   243→        .replace(/\bpi\b/gi, "Math.PI")
   244→        .replace(/\be\b/g, "Math.E")
   245→        .replace(/\bsin\(/g, "Math.sin(")
   246→        .replace(/\bcos\(/g, "Math.cos(")
   247→        .replace(/\btan\(/g, "Math.tan(")
   248→        .replace(/\bsqrt\(/g, "Math.sqrt(")
   249→        .replace(/\blog\(/g, "Math.log(")
   250→        .replace(/\bexp\(/g, "Math.exp(")
   251→        .replace(/\babs\(/g, "Math.abs(")
   252→        .replace(/\bx\b/g, String(x));
   253→      // eslint-disable-next-line no-new-func
   254→      const fn = new Function("Math", `"use strict"; return (${safeExpr});`);
   255→      const result = fn(Math);
   256→      return typeof result === "number" && isFinite(result) ? result : null;
   257→    } catch {
   258→      return null;
   259→    }
   260→  };
   261→
   262→  const points: Array<{ x: number; y: number }> = [];
   263→  const samples = 100;
   264→  for (let i = 0; i <= samples; i++) {
   265→    const x = xRange[0] + ((xRange[1] - xRange[0]) * i) / samples;
   266→    const y = evaluate(x);
   267→    if (y !== null) points.push({ x, y });
   268→  }
   269→
   270→  const toSvgX = (x: number) => padding + ((x - xRange[0]) / (xRange[1] - xRange[0])) * (width - 2 * padding);
   271→  const toSvgY = (y: number) =>
   272→    height - padding - ((y - yRange[0]) / (yRange[1] - yRange[0])) * (height - 2 * padding);
   273→
   274→  const path = points
   275→    .filter((p) => p.y >= yRange[0] && p.y <= yRange[1])
   276→    .map((p, i) => `${i === 0 ? "M" : "L"} ${toSvgX(p.x).toFixed(2)} ${toSvgY(p.y).toFixed(2)}`)
   277→    .join(" ");
   278→
   279→  const xAxisY = yRange[0] <= 0 && yRange[1] >= 0 ? toSvgY(0) : height - padding;
   280→  const yAxisX = xRange[0] <= 0 && xRange[1] >= 0 ? toSvgX(0) : padding;
   281→
   282→  const xTicks: ReactElement[] = [];
   283→  const tickCount = 5;
   284→  for (let i = 0; i <= tickCount; i++) {
   285→    const xv = xRange[0] + ((xRange[1] - xRange[0]) * i) / tickCount;
   286→    xTicks.push(
   287→      <g key={`xt-${i}`}>
   288→        <line x1={toSvgX(xv)} y1={xAxisY - 4} x2={toSvgX(xv)} y2={xAxisY + 4} stroke="#9CA3AF" strokeWidth={1} />
   289→        <text x={toSvgX(xv)} y={xAxisY + 16} fontSize={10} fill="#6B7280" textAnchor="middle">
   290→          {xv.toFixed(1)}
   291→        </text>
   292→      </g>
   293→    );
   294→  }
   295→
   296→  const yTicks: ReactElement[] = [];
   297→  for (let i = 0; i <= tickCount; i++) {
   298→    const yv = yRange[0] + ((yRange[1] - yRange[0]) * i) / tickCount;
   299→    yTicks.push(
   300→      <g key={`yt-${i}`}>
   301→        <line x1={yAxisX - 4} y1={toSvgY(yv)} x2={yAxisX + 4} y2={toSvgY(yv)} stroke="#9CA3AF" strokeWidth={1} />
   302→        <text x={yAxisX - 8} y={toSvgY(yv) + 3} fontSize={10} fill="#6B7280" textAnchor="end">
   303→          {yv.toFixed(0)}
   304→        </text>
   305→      </g>
   306→    );
   307→  }
   308→
   309→  return (
   310→    <div>
   311→      <p className="text-xs font-semibold text-gray-700 mb-1">{title}</p>
   312→      <svg viewBox={`0 0 ${width} ${height}`} className="w-full h-auto bg-gray-50 rounded-lg">
   313→        {xTicks}
   314→        {yTicks}
   315→        <line x1={padding} y1={xAxisY} x2={width - padding} y2={xAxisY} stroke="#374151" strokeWidth={1.5} />
   316→        <line x1={yAxisX} y1={padding} x2={yAxisX} y2={height - padding} stroke="#374151" strokeWidth={1.5} />
   317→        {path && <path d={path} fill="none" stroke="#4F46E5" strokeWidth={2.5} strokeLinecap="round" strokeLinejoin="round" />}
   318→        <text x={width / 2} y={height - 5} fontSize={11} fill="#374151" textAnchor="middle" fontWeight={600}>{xLabel}</text>
   319→        <text x={12} y={height / 2} fontSize={11} fill="#374151" textAnchor="middle" fontWeight={600} transform={`rotate(-90, 12, ${height / 2})`}>{yLabel}</text>
   320→      </svg>
   321→    </div>
   322→  );
   323→}

   325→// =====================================================================
   326→// 2. Scatter plot + optional line of best fit (DATA POINTS)
   327→// This directly addresses the user's physics velocity-time request.
   328→// =====================================================================
   329→function ScatterSVG({ spec }: { spec: any }) {
   330→  const title = spec.title ?? "Scatter Plot";
   331→  const xLabel = spec.xLabel ?? "x";
   332→  const yLabel = spec.yLabel ?? "y";
   333→  const points: Array<[number, number]> = Array.isArray(spec.points) ? spec.points : [];
   334→  const showBestFit = spec.lineOfBestFit !== false;
   335→
   336→  // Compute ranges from data (with optional override)
   337→  const allX = points.map((p) => p[0]);
   338→  const allY = points.map((p) => p[1]);
   339→  let xMin = spec.xRange?.[0] ?? Math.min(...allX);
   340→  let xMax = spec.xRange?.[1] ?? Math.max(...allX);
   341→  let yMin = spec.yRange?.[0] ?? Math.min(...allY);
   342→  let yMax = spec.yRange?.[1] ?? Math.max(...allY);
   343→  // Padding
   344→  const xPad = (xMax - xMin) * 0.1 || 1;
   345→  const yPad = (yMax - yMin) * 0.1 || 1;
   346→  if (!spec.xRange) { xMin -= xPad; xMax += xPad; }
   347→  if (!spec.yRange) { yMin -= yPad; yMax += yPad; }
   348→  if (yMin > 0 && spec.yRange?.[0] === undefined) yMin = 0;
   349→  if (xMin > 0 && spec.xRange?.[0] === undefined) xMin = 0;
   350→
   351→  const width = 480;
   352→  const height = 320;
   353→  const padding = 50;
   354→
   355→  const toSvgX = (x: number) => padding + ((x - xMin) / (xMax - xMin)) * (width - 2 * padding);
   356→  const toSvgY = (y: number) =>
   357→    height - padding - ((y - yMin) / (yMax - yMin)) * (height - 2 * padding);
   358→
   359→  // Linear regression for best fit line: y = mx + b
   360→  let bestFitPath: string | null = null;
   361→  let slope = 0;
   362→  let intercept = 0;
   363→  if (showBestFit && points.length >= 2) {
   364→    const n = points.length;
   365→    const sumX = allX.reduce((a, b) => a + b, 0);
   366→    const sumY = allY.reduce((a, b) => a + b, 0);
   367→    const sumXY = points.reduce((s, p) => s + p[0] * p[1], 0);
   368→    const sumXX = allX.reduce((s, x) => s + x * x, 0);
   369→    const denom = n * sumXX - sumX * sumX;
   370→    if (denom !== 0) {
   371→      slope = (n * sumXY - sumX * sumY) / denom;
   372→      intercept = (sumY - slope * sumX) / n;
   373→      const yAtXMin = slope * xMin + intercept;
   374→      const yAtXMax = slope * xMax + intercept;
   375→      bestFitPath = `M ${toSvgX(xMin).toFixed(2)} ${toSvgY(yAtXMin).toFixed(2)} L ${toSvgX(xMax).toFixed(2)} ${toSvgY(yAtXMax).toFixed(2)}`;
   376→    }
   377→  }
   378→
   379→  // Ticks
   380→  const xTicks: ReactElement[] = [];
   381→  const yTicks: ReactElement[] = [];
   382→  const tickCount = 5;
   383→  for (let i = 0; i <= tickCount; i++) {
   384→    const xv = xMin + ((xMax - xMin) * i) / tickCount;
   385→    xTicks.push(
   386→      <g key={`xt-${i}`}>
   387→        <line x1={toSvgX(xv)} y1={toSvgY(yMin) - 4} x2={toSvgX(xv)} y2={toSvgY(yMin) + 4} stroke="#9CA3AF" strokeWidth={1} />
   388→        <text x={toSvgX(xv)} y={toSvgY(yMin) + 18} fontSize={10} fill="#6B7280" textAnchor="middle">
   389→          {Number.isInteger(xv) ? xv : xv.toFixed(1)}
   390→        </text>
   391→      </g>
   392→    );
   393→  }
   394→  for (let i = 0; i <= tickCount; i++) {
   395→    const yv = yMin + ((yMax - yMin) * i) / tickCount;
   396→    yTicks.push(
   397→      <g key={`yt-${i}`}>
   398→        <line x1={toSvgX(xMin) - 4} y1={toSvgY(yv)} x2={toSvgX(xMin) + 4} y2={toSvgY(yv)} stroke="#9CA3AF" strokeWidth={1} />
   399→        <text x={toSvgX(xMin) - 8} y={toSvgY(yv) + 3} fontSize={10} fill="#6B7280" textAnchor="end">
   400→          {Number.isInteger(yv) ? yv : yv.toFixed(0)}
   401→        </text>
   402→      </g>
   403→    );
   404→  }
   405→
   406→  return (
   407→    <div>
   408→      <p className="text-xs font-semibold text-gray-700 mb-1">{title}</p>
   409→      <svg viewBox={`0 0 ${width} ${height}`} className="w-full h-auto bg-gray-50 rounded-lg">
   410→        {/* Grid lines */}
   411→        {xTicks}
   412→        {yTicks}
   413→        {/* Axes */}
   414→        <line x1={toSvgX(xMin)} y1={toSvgY(yMin)} x2={toSvgX(xMax)} y2={toSvgY(yMin)} stroke="#374151" strokeWidth={1.5} />
   415→        <line x1={toSvgX(xMin)} y1={toSvgY(yMin)} x2={toSvgX(xMin)} y2={toSvgY(yMax)} stroke="#374151" strokeWidth={1.5} />
   416→        {/* Best fit line */}
   417→        {bestFitPath && <path d={bestFitPath} fill="none" stroke="#EF4444" strokeWidth={2} strokeDasharray="6 4" opacity={0.85} />}
   418→        {/* Data points */}
   419→        {/* Data points — interactive (hover for tooltip with values) */}
   420→        {points.map((p, i) => (
   421→          <HoverPoint
   422→            key={`pt-${i}`}
   423→            cx={toSvgX(p[0])}
   424→            cy={toSvgY(p[1])}
   425→            label={`(${p[0]}, ${p[1]})`}
   426→            color="#4F46E5"
   427→          />
   428→        ))}
   429→        {/* Labels */}
   430→        <text x={width / 2} y={height - 8} fontSize={11} fill="#374151" textAnchor="middle" fontWeight={600}>{xLabel}</text>
   431→        <text x={14} y={height / 2} fontSize={11} fill="#374151" textAnchor="middle" fontWeight={600} transform={`rotate(-90, 14, ${height / 2})`}>{yLabel}</text>
   432→      </svg>
   433→      {showBestFit && bestFitPath && (
   434→        <p className="text-[10px] text-gray-500 mt-1">
   435→          Line of best fit: <span className="font-mono font-semibold">y = {slope.toFixed(3)}x {intercept >= 0 ? "+" : "−"} {Math.abs(intercept).toFixed(3)}</span>
   436→        </p>
   437→      )}
   438→    </div>
   439→  );
   440→}

   442→// =====================================================================
   443→// 3. Bar chart (vertical bars with categories)
   444→// =====================================================================
   445→function BarChartSVG({ spec }: { spec: any }) {
   446→  const title = spec.title ?? "Bar Chart";
   447→  const xLabel = spec.xLabel ?? "";
   448→  const yLabel = spec.yLabel ?? "";
   449→  const categories: string[] = Array.isArray(spec.categories) ? spec.categories : [];
   450→  const values: number[] = Array.isArray(spec.values) ? spec.values : [];
   451→  const colors: string[] = Array.isArray(spec.colors) ? spec.colors : [];
   452→
   453→  const width = 480;
   454→  const height = 320;
   455→  const padding = { left: 50, right: 20, top: 30, bottom: 50 };
   456→  const plotW = width - padding.left - padding.right;
   457→  const plotH = height - padding.top - padding.bottom;
   458→
   459→  const yMax = Math.max(...values, 0) * 1.1 || 10;
   460→  const yMin = Math.min(...values, 0);
   461→  const yRange = yMax - yMin || 1;
   462→
   463→  const toSvgY = (v: number) => padding.top + plotH - ((v - yMin) / yRange) * plotH;
   464→
   465→  const barWidth = plotW / Math.max(categories.length, 1);
   466→  const barInnerWidth = barWidth * 0.65;
   467→
   468→  // Y ticks
   469→  const yTicks: ReactElement[] = [];
   470→  const tickCount = 5;
   471→  for (let i = 0; i <= tickCount; i++) {
   472→    const yv = yMin + (yRange * i) / tickCount;
   473→    yTicks.push(
   474→      <g key={`yt-${i}`}>
   475→        <line x1={padding.left - 4} y1={toSvgY(yv)} x2={padding.left + plotW} y2={toSvgY(yv)} stroke="#E5E7EB" strokeWidth={1} />
   476→        <text x={padding.left - 8} y={toSvgY(yv) + 3} fontSize={10} fill="#6B7280" textAnchor="end">
   477→          {yv.toFixed(0)}
   478→        </text>
   479→      </g>
   480→    );

   592→function PieChartSVG({ spec }: { spec: any }) {
   593→  const title = spec.title ?? "Pie Chart";
   594→  const slices: Array<{ label: string; value: number; color?: string }> = Array.isArray(spec.slices)
   595→    ? spec.slices
   596→    : [];
   597→  const total = slices.reduce((s, sl) => s + (sl.value ?? 0), 0) || 1;
   598→  const width = 420;
   599→  const height = 320;
   600→  const cx = 160;
   601→  const cy = 160;
   602→  const r = 110;
   603→
   604→  let cumulativeAngle = -Math.PI / 2; // start at 12 o'clock
   605→  const arcs: ReactElement[] = [];
   606→  const legend: ReactElement[] = [];
   607→  slices.forEach((sl, i) => {
   608→    const angle = (sl.value / total) * 2 * Math.PI;
   609→    const x1 = cx + r * Math.cos(cumulativeAngle);
   610→    const y1 = cy + r * Math.sin(cumulativeAngle);
   611→    const x2 = cx + r * Math.cos(cumulativeAngle + angle);
   612→    const y2 = cy + r * Math.sin(cumulativeAngle + angle);
   613→    const largeArc = angle > Math.PI ? 1 : 0;
   614→    const color = sl.color ?? PALETTE[i % PALETTE.length];
   615→    const path = `M ${cx} ${cy} L ${x1.toFixed(2)} ${y1.toFixed(2)} A ${r} ${r} 0 ${largeArc} 1 ${x2.toFixed(2)} ${y2.toFixed(2)} Z`;
   616→    arcs.push(
   617→      <path key={`arc-${i}`} d={path} fill={color} stroke="white" strokeWidth={2} opacity={0.9} />
   618→    );
   619→    // Percentage label at slice center
   620→    const midAngle = cumulativeAngle + angle / 2;
   621→    const labelX = cx + (r * 0.65) * Math.cos(midAngle);
   622→    const labelY = cy + (r * 0.65) * Math.sin(midAngle);
   623→    const pct = ((sl.value / total) * 100).toFixed(1);
   624→    arcs.push(
   625→      <text key={`pct-${i}`} x={labelX} y={labelY + 4} fontSize={11} fill="white" textAnchor="middle" fontWeight={700}>
   626→        {pct}%
   627→      </text>
   628→    );
   629→    // Legend
   630→    legend.push(
   631→      <div key={`leg-${i}`} className="flex items-center gap-1.5 text-[10px]">
   632→        <span style={{ background: color, width: 10, height: 10, borderRadius: 2, display: "inline-block" }} />
   633→        <span className="text-gray-700">{sl.label} ({pct}%)</span>
   634→      </div>
   635→    );
   636→    cumulativeAngle += angle;
   637→  });
   638→
   639→  return (
   640→    <div>
   641→      <p className="text-xs font-semibold text-gray-700 mb-1">{title}</p>
   642→      <div className="flex flex-col sm:flex-row gap-2">
   643→        <svg viewBox={`0 0 ${width} ${height}`} className="w-full sm:w-1/2 h-auto bg-gray-50 rounded-lg">
   644→          {arcs}
   645→          <text x={cx} y={cy + 4} fontSize={11} fill="#374151" textAnchor="middle" fontWeight={700}>Total: {total}</text>
   646→        </svg>
   647→        <div className="flex-1 space-y-1">{legend}</div>
   648→      </div>
   649→    </div>
   650→  );

   803→// 8. Tree diagram (probability)
   804→// =====================================================================
   805→type TreeNode = {
   806→  label: string;
   807→  probability?: string;
   808→  children?: TreeNode[];
   809→};
   810→
   811→function TreeSVG({ spec }: { spec: any }) {
   812→  const title = spec.title ?? "Tree Diagram";
   813→  const root: TreeNode = spec.root ?? { label: "Start" };
   814→
   815→  const width = 600;
   816→  const height = 380;
   817→  const padding = 30;
   818→
   819→  // Layout: BFS to assign positions
   820→  // depth 0 = root at left
   821→  // each level adds more nodes to the right
   822→  type LayoutNode = {
   823→    node: TreeNode;
   824→    x: number;
   825→    y: number;
   826→    depth: number;
   827→    parent?: { x: number; y: number; probability?: string };
   828→  };
   829→
   830→  const layoutNodes: LayoutNode[] = [];
   831→  let yCounter = 0;
   832→  const maxDepth = 3;
   833→
   834→  function place(node: TreeNode, depth: number, parent?: { x: number; y: number; probability?: string }) {
   835→    if (depth > maxDepth) return;
   836→    const x = padding + depth * 180;
   837→    if (!node.children || node.children.length === 0) {
   838→      const y = padding + yCounter * 36;
   839→      yCounter++;
   840→      layoutNodes.push({ node, x, y, depth, parent });
   841→    } else {
   842→      // first compute children positions to find midpoint
   843→      const childYs: number[] = [];
   844→      for (const child of node.children) {
   845→        const before = yCounter;
   846→        place(child, depth + 1, { x, y: 0, probability: child.probability });
   847→        // after recursive call, find child y — it's the last leaf pushed
   848→        // Take the average of new leaves
   849→        const after = yCounter;
   850→        const midY = (before + after - 1) / 2;
   851→        childYs.push(padding + midY * 36);
   852→      }
   853→      const midY = childYs.reduce((a, b) => a + b, 0) / childYs.length;
   854→      layoutNodes.push({ node, x, y: midY, depth, parent });
   855→    }
   856→  }
   857→
   858→  // Simpler approach: do it iteratively for ≤3 levels
   859→  // Layout children of a node vertically centered around parent.y
   860→  function layout(node: TreeNode, depth: number, parentY: number, parentX: number, probability?: string) {
   861→    if (depth > maxDepth) return;
   862→    const x = padding + depth * 170;
   863→    if (!node.children || node.children.length === 0) {
   864→      layoutNodes.push({ node, x, y: parentY, depth, parent: parentX > 0 ? { x: parentX, y: parentY, probability } : undefined });
   865→      return;
   866→    }
   867→    const childCount = node.children.length;
   868→    const totalHeight = childCount * 50;
   869→    const startY = parentY - totalHeight / 2 + 25;
   870→    const childYs: number[] = [];
   871→    node.children.forEach((child, i) => {
   872→      const cy = startY + i * 50;
   873→      childYs.push(cy);
   874→    });
   875→    const myY = childYs.reduce((a, b) => a + b, 0) / childYs.length;
   876→    layoutNodes.push({ node, x, y: myY, depth, parent: parentX > 0 ? { x: parentX, y: parentY, probability } : undefined });
   877→    node.children.forEach((child, i) => {
   878→      layout(child, depth + 1, childYs[i], x, child.probability);
   879→    });
   880→  }
   881→
   882→  layout(root, 0, height / 2, 0);
   883→
   884→  return (
   885→    <div>
   886→      <p className="text-xs font-semibold text-gray-700 mb-1">{title}</p>
   887→      <svg viewBox={`0 0 ${width} ${height}`} className="w-full h-auto bg-gray-50 rounded-lg">
   888→        {/* Edges */}
   889→        {layoutNodes.map((ln, i) => {
   890→          if (!ln.parent) return null;
   891→          return (
   892→            <g key={`edge-${i}`}>
   893→              <line
   894→                x1={ln.parent.x}
   895→                y1={ln.parent.y}
   896→                x2={ln.x}
   897→                y2={ln.y}
   898→                stroke="#9CA3AF"
   899→                strokeWidth={1.5}
   900→              />
   901→              {ln.parent.probability && (
   902→                <text
   903→                  x={(ln.parent.x + ln.x) / 2}
   904→                  y={(ln.parent.y + ln.y) / 2 - 4}
   905→                  fontSize={10}
   906→                  fill="#6B7280"
   907→                  textAnchor="middle"
   908→                  fontWeight={600}
   909→                >
   910→                  {ln.parent.probability}
   911→                </text>
   912→              )}
   913→            </g>
   914→          );
   915→        })}
   916→        {/* Nodes */}
   917→        {layoutNodes.map((ln, i) => (
   918→          <g key={`node-${i}`}>
   919→            <rect
   920→              x={ln.x - 30}
   921→              y={ln.y - 12}
   922→              width={60}
   923→              height={24}
   924→              rx={12}
   925→              fill={PALETTE[ln.depth % PALETTE.length]}
   926→              opacity={0.9}
   927→            />
   928→            <text x={ln.x} y={ln.y + 4} fontSize={11} fill="white" textAnchor="middle" fontWeight={700}>
   929→              {ln.node.label}
   930→            </text>
   931→          </g>
   932→        ))}
   933→      </svg>
   934→    </div>
   935→  );
   936→}

   938→// =====================================================================
   939→// 9. Network graph (also useful for graph theory / social networks)
   940→// =====================================================================
   941→function NetworkSVG({ spec }: { spec: any }) {
   942→  const title = spec.title ?? "Network Graph";
   943→  const nodes: Array<{ id: string; label: string; color?: string }> = Array.isArray(spec.nodes) ? spec.nodes : [];
   944→  const edges: Array<{ from: string; to: string; label?: string; directed?: boolean }> = Array.isArray(spec.edges) ? spec.edges : [];
   945→
   946→  const width = 480;
   947→  const height = 360;
   948→  const cx = width / 2;
   949→  const cy = height / 2;
   950→  const radius = Math.min(width, height) / 2 - 70;
   951→
   952→  // Detect hub-and-spoke layout
   953→  const nodeDegree = new Map<string, number>();
   954→  for (const e of edges) {
   955→    nodeDegree.set(e.from, (nodeDegree.get(e.from) ?? 0) + 1);
   956→    nodeDegree.set(e.to, (nodeDegree.get(e.to) ?? 0) + 1);
   957→  }
   958→  let hubId: string | null = null;
   959→  if (nodes.length >= 4) {
   960→    const sortedByDegree = [...nodeDegree.entries()].sort((a, b) => b[1] - a[1]);
   961→    if (sortedByDegree.length > 0 && sortedByDegree[0][1] >= 3) {
   962→      const topDegree = sortedByDegree[0][1];
   963→      const secondDegree = sortedByDegree[1]?.[1] ?? 0;
   964→      if (topDegree >= secondDegree * 2) {
   965→        hubId = sortedByDegree[0][0];
   966→      }
   967→    }
   968→  }
   969→
   970→  // Compute INITIAL positions (auto-layout) — stored in state so user can drag
   971→  const computeInitial = useCallback(() => {
   972→    let pos: Array<{ x: number; y: number }>;
   973→    if (hubId) {
   974→      const hubIdx = nodes.findIndex((n) => n.id === hubId);
   975→      const otherCount = nodes.length - 1;
   976→      const angleStep = (2 * Math.PI) / Math.max(otherCount, 1);
   977→      const startAngle = -Math.PI / 2;
   978→      pos = nodes.map((n) => ({ x: cx, y: cy }));
   979→      let otherIdx = 0;
   980→      for (let i = 0; i < nodes.length; i++) {
   981→        if (i === hubIdx) {
   982→          pos[i] = { x: cx, y: cy };
   983→        } else {
   984→          const angle = startAngle + otherIdx * angleStep;
   985→          pos[i] = {
   986→            x: cx + radius * Math.cos(angle),
   987→            y: cy + radius * Math.sin(angle),
   988→          };
   989→          otherIdx++;
   990→        }
   991→      }
   992→    } else {
   993→      pos = nodes.map((_, i) => {
   994→        const angle = (2 * Math.PI * i) / Math.max(nodes.length, 1) - Math.PI / 2;
   995→        return {
   996→          x: cx + radius * Math.cos(angle),
   997→          y: cy + radius * Math.sin(angle),
   998→        };
   999→      });
  1000→    }
  1001→    return pos;
  1002→  }, [hubId, nodes, cx, cy, radius]);
  1003→
  1004→  const [positions, setPositions] = useState<Array<{ x: number; y: number }>>(computeInitial);
  1005→  // Reset positions when spec changes (new conversation)
  1006→  useEffect(() => {
  1007→    setPositions(computeInitial());
  1008→  }, [computeInitial]);
  1009→
  1010→  // Drag state
  1011→  const [dragging, setDragging] = useState<string | null>(null);
  1012→  const svgRef = useRef<SVGSVGElement | null>(null);
  1013→
  1014→  const onNodeMouseDown = (id: string, e: React.MouseEvent) => {
  1015→    e.preventDefault();
  1016→    setDragging(id);
  1017→  };
  1018→
  1019→  const onMouseMove = (e: React.MouseEvent) => {
  1020→    if (!dragging || !svgRef.current) return;
  1021→    const rect = svgRef.current.getBoundingClientRect();
  1022→    // Convert mouse coords to SVG viewBox coords
  1023→    const scale = width / rect.width;
  1024→    const svgX = (e.clientX - rect.left) * scale;
  1025→    const svgY = (e.clientY - rect.top) * (height / rect.height);
  1026→    setPositions((prev) =>
  1027→      prev.map((p, i) => (nodes[i]?.id === dragging ? { x: svgX, y: svgY } : p))
  1028→    );
  1029→  };
  1030→
  1031→  const onMouseUp = () => setDragging(null);
  1032→
  1033→  return (
  1034→    <div>
  1035→      {title && <p className="text-xs font-semibold text-gray-700 mb-1">{title}</p>}
  1036→      <svg
  1037→        viewBox={`0 0 ${width} ${height}`}
  1038→        className="w-full h-auto bg-gray-50 rounded-lg select-none"
  1039→        ref={svgRef}
  1040→        onMouseMove={onMouseMove}
  1041→        onMouseUp={onMouseUp}
  1042→        onMouseLeave={onMouseUp}
  1043→        style={{ touchAction: "none" }}
  1044→      >
  1045→        {/* Edges */}
  1046→        {edges.map((edge, i) => {
  1047→          const fromIdx = nodes.findIndex((n) => n.id === edge.from);
  1048→          const toIdx = nodes.findIndex((n) => n.id === edge.to);
  1049→          if (fromIdx < 0 || toIdx < 0) return null;
  1050→          const from = positions[fromIdx];
  1051→          const to = positions[toIdx];
  1052→          const midX = (from.x + to.x) / 2;
  1053→          const midY = (from.y + to.y) / 2;
  1054→          return (
  1055→            <g key={`edge-${i}`}>
  1056→              <line x1={from.x} y1={from.y} x2={to.x} y2={to.y} stroke="#9CA3AF" strokeWidth={1.5} />
  1057→              {edge.directed && (
  1058→                <polygon
  1059→                  points={arrowPoints(from.x, from.y, to.x, to.y, 8)}
  1060→                  fill="#9CA3AF"
  1061→                />
  1062→              )}
  1063→              {edge.label && (
  1064→                <text x={midX} y={midY - 4} fontSize={10} fill="#6B7280" textAnchor="middle" fontWeight={600}>
  1065→                  {edge.label}
  1066→                </text>
  1067→              )}
  1068→            </g>
  1069→          );
  1070→        })}
  1071→        {/* Nodes (draggable) */}
  1072→        {nodes.map((n, i) => {
  1073→          const pos = positions[i];
  1074→          const isHub = n.id === hubId;
  1075→          const isDragging = dragging === n.id;
  1076→          const color = n.color ?? PALETTE[i % PALETTE.length];
  1077→          const fontSize = isHub ? 12 : 11;
  1078→          const padding = isHub ? 12 : 8;
  1079→          const rectHeight = isHub ? 36 : 32;
  1080→          const labelWidth = Math.max(isHub ? 100 : 60, n.label.length * 7 + padding);
  1081→          return (
  1082→            <g
  1083→              key={n.id}
  1084→              onMouseDown={(e) => onNodeMouseDown(n.id, e)}
  1085→              style={{ cursor: isDragging ? "grabbing" : "grab" }}
  1086→            >
  1087→              {isHub && (
  1088→                // Glow / highlight ring around the hub
  1089→                <rect
  1090→                  x={pos.x - labelWidth / 2 - 3}
  1091→                  y={pos.y - rectHeight / 2 - 3}
  1092→                  width={labelWidth + 6}
  1093→                  height={rectHeight + 6}
  1094→                  rx={(rectHeight + 6) / 2}
  1095→                  fill="none"
  1096→                  stroke={color}
  1097→                  strokeWidth={1.5}
  1098→                  strokeOpacity={0.4}
  1099→                />
  1100→              )}
  1101→              <rect
  1102→                x={pos.x - labelWidth / 2}
  1103→                y={pos.y - rectHeight / 2}
  1104→                width={labelWidth}
  1105→                height={rectHeight}
  1106→                rx={rectHeight / 2}
  1107→                fill={color}
  1108→                opacity={0.95}
  1109→              />
  1110→              <text
  1111→                x={pos.x}
  1112→                y={pos.y + 4}
  1113→                fontSize={fontSize}
  1114→                fill="white"
  1115→                textAnchor="middle"
  1116→                fontWeight={isHub ? 700 : 600}
  1117→                style={{ pointerEvents: "none" }}
  1118→              >
  1119→                {n.label.length > (isHub ? 28 : 22) ? n.label.slice(0, isHub ? 28 : 22) + "…" : n.label}
  1120→              </text>
  1121→            </g>
  1122→          );
  1123→        })}
  1124→      </svg>
  1125→      <p className="text-[10px] text-gray-400 mt-1 flex items-center gap-1">
  1126→        <span className="cursor-grab">✋</span> Drag nodes to rearrange
  1127→      </p>
  1128→    </div>
  1129→  );
  1130→}

  1132→function arrowPoints(fromX: number, fromY: number, toX: number, toY: number, size: number): string {
  1133→  // shorten endpoint to node radius
  1134→  const dx = toX - fromX;
  1135→  const dy = toY - fromY;
  1136→  const len = Math.sqrt(dx * dx + dy * dy) || 1;
  1137→  const nx = dx / len;
  1138→  const ny = dy / len;
  1139→  const tipX = toX - nx * 18;
  1140→  const tipY = toY - ny * 18;
  1141→  const baseX = tipX - nx * size;
  1142→  const baseY = tipY - ny * size;
  1143→  const perpX = -ny * size * 0.5;
  1144→  const perpY = nx * size * 0.5;
  1145→  return `${tipX},${tipY} ${baseX + perpX},${baseY + perpY} ${baseX - perpX},${baseY - perpY}`;
  1146→}

  1224→function PolygonSVG({ spec }: { spec: any }) {
  1225→  const title = spec.title ?? "Polygon";
  1226→  const vertices: Array<[number, number]> = Array.isArray(spec.vertices) ? spec.vertices : [];
  1227→  const labels: string[] = Array.isArray(spec.labels) ? spec.labels : [];
  1228→  const showAngles = spec.showAngles !== false;
  1229→  const showSides = spec.showSides !== false;
  1230→
  1231→  // Compute bounds
  1232→  const allX = vertices.map((v) => v[0]);
  1233→  const allY = vertices.map((v) => v[1]);
  1234→  let xMin = Math.min(...allX);
  1235→  let xMax = Math.max(...allX);
  1236→  let yMin = Math.min(...allY);
  1237→  let yMax = Math.max(...allY);
  1238→  const xPad = (xMax - xMin) * 0.2 || 1;
  1239→  const yPad = (yMax - yMin) * 0.2 || 1;
  1240→  xMin -= xPad; xMax += xPad; yMin -= yPad; yMax += yPad;
  1241→
  1242→  const width = 380;
  1243→  const height = 320;
  1244→  const padding = 30;
  1245→
  1246→  const toSvgX = (x: number) => padding + ((x - xMin) / (xMax - xMin)) * (width - 2 * padding);
  1247→  const toSvgY = (y: number) =>
  1248→    height - padding - ((y - yMin) / (yMax - yMin)) * (height - 2 * padding);
  1249→
  1250→  const path = vertices.map((v, i) => `${i === 0 ? "M" : "L"} ${toSvgX(v[0]).toFixed(2)} ${toSvgY(v[1]).toFixed(2)}`).join(" ") + " Z";
  1251→
  1252→  // Side lengths
  1253→  const sides: Array<{ length: number; midX: number; midY: number }> = [];
  1254→  for (let i = 0; i < vertices.length; i++) {
  1255→    const a = vertices[i];
  1256→    const b = vertices[(i + 1) % vertices.length];
  1257→    const len = Math.sqrt((b[0] - a[0]) ** 2 + (b[1] - a[1]) ** 2);
  1258→    sides.push({
  1259→      length: len,
  1260→      midX: toSvgX((a[0] + b[0]) / 2),
  1261→      midY: toSvgY((a[1] + b[1]) / 2),
  1262→    });
  1263→  }
  1264→
  1265→  return (
  1266→    <div>
  1267→      <p className="text-xs font-semibold text-gray-700 mb-1">{title}</p>
  1268→      <svg viewBox={`0 0 ${width} ${height}`} className="w-full h-auto bg-gray-50 rounded-lg">
  1269→        {/* Polygon */}
  1270→        <path d={path} fill="#4F46E5" fillOpacity={0.15} stroke="#4F46E5" strokeWidth={2.5} />
  1271→        {/* Vertices + labels */}
  1272→        {vertices.map((v, i) => {
  1273→          const label = labels[i] ?? String.fromCharCode(65 + i);
  1274→          // Offset label outward (from centroid)
  1275→          const cx = vertices.reduce((s, p) => s + p[0], 0) / vertices.length;
  1276→          const cy = vertices.reduce((s, p) => s + p[1], 0) / vertices.length;
  1277→          const dx = v[0] - cx;
  1278→          const dy = v[1] - cy;
  1279→          const len = Math.sqrt(dx * dx + dy * dy) || 1;
  1280→          const labelOffset = 16;
  1281→          const labelX = toSvgX(v[0]) + (dx / len) * labelOffset;
  1282→          const labelY = toSvgY(v[1]) + (dy / len) * labelOffset;
  1283→          return (
  1284→            <g key={`v-${i}`}>
  1285→              <circle cx={toSvgX(v[0])} cy={toSvgY(v[1])} r={3.5} fill="#4F46E5" />
  1286→              <text x={labelX} y={labelY + 4} fontSize={13} fill="#374151" textAnchor="middle" fontWeight={700}>
  1287→                {label}
  1288→              </text>
  1289→            </g>
  1290→          );
  1291→        })}
  1292→        {/* Side length labels */}
  1293→        {showSides && sides.map((s, i) => (
  1294→          <text key={`s-${i}`} x={s.midX} y={s.midY - 4} fontSize={10} fill="#10B981" textAnchor="middle" fontWeight={600}>
  1295→            {s.length.toFixed(2)}
  1296→          </text>
  1297→        ))}
  1298→      </svg>
  1299→    </div>
  1300→  );
  1301→}

  1404→function SlopeFieldSVG({ spec }: { spec: any }) {

  1406→  const expr = spec.expr ?? "x"; // dy/dx = expr (function of x and y)
  1407→  const xRange: [number, number] = spec.xRange ?? [-5, 5];
  1408→  const yRange: [number, number] = spec.yRange ?? [-5, 5];
  1409→  const gridSize: number = spec.gridSize ?? 10; // grid points per axis
  1410→  const xLabel = spec.xLabel ?? "x";
  1411→  const yLabel = spec.yLabel ?? "y";
  1412→  const width = 480;
  1413→  const height = 360;
  1414→  const padding = 40;
  1415→
  1416→  // Safe expression evaluator — supports Math.* + x + y
  1417→  const evaluate = (x: number, y: number): number | null => {
  1418→    try {
  1419→      let safeExpr = expr
  1420→        .replace(/\^/g, "**")
  1421→        .replace(/\bpi\b/gi, "Math.PI")
  1422→        .replace(/\be\b/g, "Math.E")
  1423→        .replace(/\bsin\(/g, "Math.sin(")
  1424→        .replace(/\bcos\(/g, "Math.cos(")
  1425→        .replace(/\btan\(/g, "Math.tan(")
  1426→        .replace(/\bsqrt\(/g, "Math.sqrt(")
  1427→        .replace(/\blog\(/g, "Math.log(")
  1428→        .replace(/\bexp\(/g, "Math.exp(")
  1429→        .replace(/\babs\(/g, "Math.abs(")
  1430→        .replace(/\bx\b/g, String(x))
  1431→        .replace(/\by\b/g, String(y));
  1432→      // eslint-disable-next-line no-new-func
  1433→      const fn = new Function("Math", `"use strict"; return (${safeExpr});`);
  1434→      const result = fn(Math);
  1435→      return typeof result === "number" && isFinite(result) ? result : null;
  1436→    } catch {
  1437→      return null;
  1438→    }
  1439→  };
  1440→
  1441→  const toSvgX = (x: number) => padding + ((x - xRange[0]) / (xRange[1] - xRange[0])) * (width - 2 * padding);
  1442→  const toSvgY = (y: number) =>
  1443→    height - padding - ((y - yRange[0]) / (yRange[1] - yRange[0])) * (height - 2 * padding);
  1444→
  1445→  // Generate slope tick marks
  1446→  const ticks: ReactElement[] = [];
  1447→  const tickLength = 8;
  1448→  for (let i = 0; i <= gridSize; i++) {
  1449→    for (let j = 0; j <= gridSize; j++) {
  1450→      const x = xRange[0] + ((xRange[1] - xRange[0]) * i) / gridSize;
  1451→      const y = yRange[0] + ((yRange[1] - yRange[0]) * j) / gridSize;
  1452→      const slope = evaluate(x, y);
  1453→      if (slope === null) continue;
  1454→      // Normalize direction
  1455→      const dx = 1 / Math.sqrt(1 + slope * slope);
  1456→      const dy = slope / Math.sqrt(1 + slope * slope);
  1457→      const cx = toSvgX(x);
  1458→      const cy = toSvgY(y);
  1459→      const x1 = cx - (tickLength / 2) * dx;
  1460→      const y1 = cy + (tickLength / 2) * dy; // SVG y inverted
  1461→      const x2 = cx + (tickLength / 2) * dx;
  1462→      const y2 = cy - (tickLength / 2) * dy;
  1463→      ticks.push(
  1464→        <line
  1465→          key={`tick-${i}-${j}`}
  1466→          x1={x1.toFixed(2)}
  1467→          y1={y1.toFixed(2)}
  1468→          x2={x2.toFixed(2)}
  1469→          y2={y2.toFixed(2)}
  1470→          stroke="#4F46E5"
  1471→          strokeWidth={1.5}
  1472→          opacity={0.7}
  1473→        />
  1474→      );
  1475→    }
  1476→  }
  1477→
  1478→  // Axes
  1479→  const xAxisY = yRange[0] <= 0 && yRange[1] >= 0 ? toSvgY(0) : height - padding;
  1480→  const yAxisX = xRange[0] <= 0 && xRange[1] >= 0 ? toSvgX(0) : padding;
  1481→
  1482→  return (
  1483→    <div>
  1484→      <p className="text-xs font-semibold text-gray-700 mb-1">{title}</p>
  1485→      <p className="text-[10px] text-gray-500 mb-1">dy/dx = <span className="font-mono">{expr}</span></p>
  1486→      <svg viewBox={`0 0 ${width} ${height}`} className="w-full h-auto bg-gray-50 rounded-lg">
  1487→        {ticks}
  1488→        <line x1={padding} y1={xAxisY} x2={width - padding} y2={xAxisY} stroke="#374151" strokeWidth={1.5} />
  1489→        <line x1={yAxisX} y1={padding} x2={yAxisX} y2={height - padding} stroke="#374151" strokeWidth={1.5} />
  1490→        <text x={width / 2} y={height - 8} fontSize={11} fill="#374151" textAnchor="middle" fontWeight={600}>{xLabel}</text>
  1491→        <text x={12} y={height / 2} fontSize={11} fill="#374151" textAnchor="middle" fontWeight={600} transform={`rotate(-90, 12, ${height / 2})`}>{yLabel}</text>
  1492→      </svg>
  1493→    </div>
  1494→  );
  1495→}

  1500→function StemLeafSVG({ spec }: { spec: any }) {
  1501→  const title = spec.title ?? "Stem-and-Leaf Plot";
  1502→  const data: number[] = Array.isArray(spec.data) ? spec.data : [];
  1503→  const stemUnit: number = spec.stemUnit ?? 10; // e.g. 10 means stem = tens digit
  1504→  const leafUnit: number = spec.leafUnit ?? 1; // e.g. 1 means leaf = ones digit
  1505→
  1506→  if (data.length === 0) {
  1507→    return <p className="text-xs text-gray-500">No data provided for stem-and-leaf plot.</p>;
  1508→  }
  1509→
  1510→  // Group by stem
  1511→  const groups = new Map<number, number[]>();
  1512→  for (const v of data) {
  1513→    const stem = Math.floor(v / stemUnit);
  1514→    const leaf = Math.floor((v - stem * stemUnit) / leafUnit);
  1515→    if (!groups.has(stem)) groups.set(stem, []);
  1516→    groups.get(stem)!.push(leaf);
  1517→  }
  1518→  // Sort stems
  1519→  const sortedStems = [...groups.keys()].sort((a, b) => a - b);
  1520→
  1521→  return (
  1522→    <div>
  1523→      <p className="text-xs font-semibold text-gray-700 mb-1">{title}</p>
  1524→      <p className="text-[10px] text-gray-500 mb-2">Stem = {stemUnit}s, Leaf = {leafUnit}s</p>
  1525→      <div className="bg-white border border-gray-200 rounded-lg p-3 font-mono text-xs">
  1526→        <table className="w-full">
  1527→          <thead>
  1528→            <tr className="border-b border-gray-200 text-gray-600">
  1529→              <th className="text-left py-1 px-2">Stem</th>
  1530→              <th className="text-left py-1 px-2">Leaves</th>
  1531→              <th className="text-right py-1 px-2 text-gray-400">Count</th>
  1532→            </tr>
  1533→          </thead>
  1534→          <tbody>
  1535→            {sortedStems.map((stem) => {
  1536→              const leaves = (groups.get(stem) ?? []).sort((a, b) => a - b);
  1537→              return (
  1538→                <tr key={stem} className="border-b border-gray-50">
  1539→                  <td className="py-1 px-2 font-bold text-indigo-600">{stem} |</td>
  1540→                  <td className="py-1 px-2 tracking-wide text-gray-800">{leaves.join(" ")}</td>
  1541→                  <td className="py-1 px-2 text-right text-gray-400">{leaves.length}</td>
  1542→                </tr>
  1543→              );
  1544→            })}
  1545→          </tbody>
  1546→        </table>
  1547→      </div>
  1548→      <p className="text-[10px] text-gray-500 mt-1">n = {data.length} · Key: {sortedStems[0] ?? 0}|{(groups.get(sortedStems[0]) ?? [0])[0]} = {(sortedStems[0] ?? 0) * stemUnit + (groups.get(sortedStems[0]) ?? [0])[0] * leafUnit}</p>
  1549→    </div>
  1550→  );
  1551→}

  1628→// =====================================================================
  1629→// 16. Freeform SVG — AI can output raw SVG markup for any custom drawing
  1630→//     The "svg" field in the spec contains the raw SVG body (without the
  1631→//     outer <svg> tag). We wrap it in a viewBox and render it inline.
  1632→//     DANGEROUS: this executes AI-generated SVG, so we sanitize by stripping
  1633→//     <script> tags, on* event handlers, and external resource references.
  1634→// =====================================================================
  1635→function FreeformSVG({ spec }: { spec: any }) {
  1636→  const title = spec.title ?? "Custom Drawing";
  1637→  const rawSvg: string = typeof spec.svg === "string" ? spec.svg : "";
  1638→  let width: number = spec.width ?? 480;
  1639→  let height: number = spec.height ?? 360;
  1640→
  1641→  if (!rawSvg) {
  1642→    return (
  1643→      <div className="text-xs text-rose-600 p-3">
  1644→        Freeform graph type requires an "svg" field containing raw SVG markup.
  1645→      </div>
  1646→    );
  1647→  }
  1648→
  1649→  // If the AI wrapped its content in an outer <svg>...</svg> tag, extract the
  1650→  // INNER content (we'll wrap it in our own <svg> below). This avoids the
  1651→  // nested-<svg> rendering issue where browsers don't reliably render an
  1652→  // inner SVG element placed via dangerouslySetInnerHTML.
  1653→  // Also extract the viewBox/width/height attributes from the AI's <svg> tag
  1654→  // so we can use the AI's intended dimensions.
  1655→  let svgContent = rawSvg;
  1656→  const outerSvgMatch = rawSvg.match(/<svg\b([^>]*)>([\s\S]*?)<\/svg>/i);
  1657→  if (outerSvgMatch) {
  1658→    const outerAttrs = outerSvgMatch[1] ?? "";
  1659→    svgContent = outerSvgMatch[2] ?? "";
  1660→
  1661→    // Try to extract viewBox / width / height from the AI's <svg> tag
  1662→    const viewBoxMatch = outerAttrs.match(/viewBox\s*=\s*["']([^"']+)["']/i);
  1663→    if (viewBoxMatch) {
  1664→      const parts = viewBoxMatch[1].split(/[\s,]+/).map(Number);
  1665→      if (parts.length === 4 && parts.every((p) => !isNaN(p))) {
  1666→        width = parts[2];
  1667→        height = parts[3];
  1668→      }
  1669→    }
  1670→    const widthMatch = outerAttrs.match(/\bwidth\s*=\s*["'](\d+)["']/i);
  1671→    const heightMatch = outerAttrs.match(/\bheight\s*=\s*["'](\d+)["']/i);
  1672→    if (widthMatch) width = parseInt(widthMatch[1], 10);
  1673→    if (heightMatch) height = parseInt(heightMatch[1], 10);
  1674→  } else if (spec.width && spec.height) {
  1675→    // No outer svg tag — use spec dimensions
  1676→    width = spec.width;
  1677→    height = spec.height;
  1678→  }
  1679→
  1680→  // Sanitize: strip <script>...</script>, on* handlers, javascript: URLs,
  1681→  // external image refs, and data: URLs (to prevent XSS and data exfiltration).
  1682→  let sanitized = svgContent;
  1683→  // Remove <script>...</script> blocks
  1684→  sanitized = sanitized.replace(/<script[\s\S]*?<\/script>/gi, "");
  1685→  // Remove on* event handlers (onclick, onload, onerror, etc.)
  1686→  sanitized = sanitized.replace(/\son\w+\s*=\s*"[^"]*"/gi, "");
  1687→  sanitized = sanitized.replace(/\son\w+\s*=\s*'[^']*'/gi, "");
  1688→  // Remove javascript: URLs
  1689→  sanitized = sanitized.replace(/href\s*=\s*"javascript:[^"]*"/gi, 'href="#"');
  1690→  sanitized = sanitized.replace(/href\s*=\s*'javascript:[^']*'/gi, "href='#'");
  1691→  // Remove external image references (prevent data exfiltration via image URLs)
  1692→  sanitized = sanitized.replace(/href\s*=\s*"https?:\/\/[^"]*"/gi, 'href="#"');
  1693→  sanitized = sanitized.replace(/href\s*=\s*'https?:\/\/[^']*'/gi, "href='#'");
  1694→  // Remove xlink:href external references too
  1695→  sanitized = sanitized.replace(/xlink:href\s*=\s*"https?:\/\/[^"]*"/gi, 'xlink:href="#"');
  1696→
  1697→  return (
  1698→    <div>
  1699→      {title && <p className="text-xs font-semibold text-gray-700 mb-1">{title}</p>}
  1700→      <svg
  1701→        viewBox={`0 0 ${width} ${height}`}
  1702→        className="w-full h-auto bg-gray-50 rounded-lg"
  1703→        xmlns="http://www.w3.org/2000/svg"
  1704→        dangerouslySetInnerHTML={{ __html: sanitized }}
  1705→      />
  1706→    </div>
  1707→  );
  1708→}

  1842→// 19. Vector field — arrows showing direction and magnitude at grid points
  1843→//     For F(x, y) = (P(x,y), Q(x,y)) — typically for force/magnetic fields
  1844→// =====================================================================
  1845→function VectorFieldSVG({ spec }: { spec: any }) {
  1846→  const title = spec.title ?? "Vector Field";
  1847→  // P and Q expressions (string) or precomputed vectors
  1848→  const exprP: string = spec.exprP ?? "1";
  1849→  const exprQ: string = spec.exprQ ?? "0";
  1850→  const vectors: Array<{ from?: [number, number]; to?: [number, number]; magnitude?: number }> = Array.isArray(spec.vectors) ? spec.vectors : [];
  1851→  const range: [number, number] = spec.range ?? [-5, 5];
  1852→  const gridSize: number = spec.gridSize ?? 8;
  1853→  const xLabel = spec.xLabel ?? "x";
  1854→  const yLabel = spec.yLabel ?? "y";
  1855→
  1856→  const width = 420;
  1857→  const height = 360;
  1858→  const padding = 30;
  1859→
  1860→  const toSvgX = (x: number) => padding + ((x - range[0]) / (range[1] - range[0])) * (width - 2 * padding);
  1861→  const toSvgY = (y: number) => height - padding - ((y - range[0]) / (range[1] - range[0])) * (height - 2 * padding);
  1862→
  1863→  // Evaluator for P and Q
  1864→  const eval2 = (expr: string, x: number, y: number): number | null => {
  1865→    try {
  1866→      let s = expr
  1867→        .replace(/\^/g, "**")
  1868→        .replace(/\bpi\b/gi, "Math.PI")
  1869→        .replace(/\be\b/g, "Math.E")
  1870→        .replace(/\bsin\(/g, "Math.sin(")
  1871→        .replace(/\bcos\(/g, "Math.cos(")
  1872→        .replace(/\btan\(/g, "Math.tan(")
  1873→        .replace(/\bsqrt\(/g, "Math.sqrt(")
  1874→        .replace(/\bx\b/g, String(x))
  1875→        .replace(/\by\b/g, String(y));
  1876→      // eslint-disable-next-line no-new-func
  1877→      const fn = new Function("Math", `"use strict"; return (${s});`);
  1878→      const r = fn(Math);
  1879→      return typeof r === "number" && isFinite(r) ? r : null;
  1880→    } catch { return null; }
  1881→  };
  1882→
  1883→  // If vectors are precomputed, use them; otherwise generate from exprP/exprQ
  1884→  const fieldVectors: Array<{ x: number; y: number; dx: number; dy: number; magnitude: number }> = [];
  1885→  if (vectors.length > 0) {
  1886→    for (const v of vectors) {
  1887→      const from = v.from ?? [0, 0];
  1888→      const to = v.to ?? [from[0] + 1, from[1]];
  1889→      fieldVectors.push({
  1890→        x: from[0], y: from[1],
  1891→        dx: to[0] - from[0],
  1892→        dy: to[1] - from[1],
  1893→        magnitude: v.magnitude ?? Math.sqrt((to[0] - from[0]) ** 2 + (to[1] - from[1]) ** 2),
  1894→      });
  1895→    }
  1896→  } else {
  1897→    // Generate grid of arrows
  1898→    for (let i = 0; i <= gridSize; i++) {
  1899→      for (let j = 0; j <= gridSize; j++) {
  1900→        const x = range[0] + ((range[1] - range[0]) * i) / gridSize;
  1901→        const y = range[0] + ((range[1] - range[0]) * j) / gridSize;
  1902→        const p = eval2(exprP, x, y);
  1903→        const q = eval2(exprQ, x, y);
  1904→        if (p === null || q === null) continue;
  1905→        // Normalize: arrow length = 1 unit in field direction
  1906→        const mag = Math.sqrt(p * p + q * q) || 1;
  1907→        const scale = Math.min(range[1] - range[0], range[1] - range[0]) / gridSize * 0.4;
  1908→        const dx = (p / mag) * scale;
  1909→        const dy = (q / mag) * scale;
  1910→        fieldVectors.push({ x, y, dx, dy, magnitude: mag });
  1911→      }
  1912→    }
  1913→  }
  1914→
  1915→  // Color scale by magnitude
  1916→  const maxMag = Math.max(...fieldVectors.map((v) => v.magnitude), 1);
  1917→
  1918→  return (
  1919→    <div>
  1920→      <p className="text-xs font-semibold text-gray-700 mb-1">{title}</p>
  1921→      <svg viewBox={`0 0 ${width} ${height}`} className="w-full h-auto bg-gray-50 rounded-lg">
  1922→        {/* Axes */}
  1923→        <line x1={padding} y1={toSvgY(0)} x2={width - padding} y2={toSvgY(0)} stroke="#374151" strokeWidth={1} opacity={0.5} />
  1924→        <line x1={toSvgX(0)} y1={padding} x2={toSvgX(0)} y2={height - padding} stroke="#374151" strokeWidth={1} opacity={0.5} />
  1925→        {/* Arrows */}
  1926→        {fieldVectors.map((v, i) => {
  1927→          const startX = toSvgX(v.x);
  1928→          const startY = toSvgY(v.y);
  1929→          const endX = toSvgX(v.x + v.dx);
  1930→          const endY = toSvgY(v.y + v.dy);
  1931→          const intensity = v.magnitude / maxMag;
  1932→          const color = intensity < 0.33 ? "#06B6D4" : intensity < 0.66 ? "#4F46E5" : "#EF4444";
  1933→          return (
  1934→            <g key={`vf-${i}`}>
  1935→              <line
  1936→                x1={startX}
  1937→                y1={startY}
  1938→                x2={endX}
  1939→                y2={endY}
  1940→                stroke={color}
  1941→                strokeWidth={1.5}
  1942→                opacity={0.85}
  1943→              />
  1944→              <polygon
  1945→                points={arrowPoints(startX, startY, endX, endY, 6)}
  1946→                fill={color}
  1947→              />
  1948→            </g>
  1949→          );
  1950→        })}
  1951→        <text x={width / 2} y={height - 6} fontSize={11} fill="#374151" textAnchor="middle" fontWeight={600}>{xLabel}</text>
  1952→        <text x={10} y={height / 2} fontSize={11} fill="#374151" textAnchor="middle" fontWeight={600} transform={`rotate(-90, 10, ${height / 2})`}>{yLabel}</text>
  1953→      </svg>
  1954→      <p className="text-[10px] text-gray-500 mt-1">
  1955→        {fieldVectors.length} arrows · Color: <span className="text-cyan-600">weak</span> → <span className="text-indigo-600">medium</span> → <span className="text-rose-600">strong</span>
  1956→      </p>
  1957→    </div>
  1958→  );
  1959→}
